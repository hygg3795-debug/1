return function(Window, WindUI)
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local UserInputService = game:GetService("UserInputService")
    local VirtualUser = game:GetService("VirtualUser")
    local LocalPlayer = Players.LocalPlayer

    local tab = Window:Tab({Title = "通用", Icon = "settings"})

    tab:Toggle({
        Title = "防踢",
        Value = false,
        Callback = function(val)
            if val then
                local con = LocalPlayer.Idled:Connect(function()
                    VirtualUser:CaptureController()
                    VirtualUser:ClickButton2(Vector2.new())
                end)
                _G.AntiAFKCon = con
                WindUI:Notify({Title = "防踢", Content = "已开启", Duration = 2, Icon = "check"})
            else
                if _G.AntiAFKCon then
                    _G.AntiAFKCon:Disconnect()
                    _G.AntiAFKCon = nil
                end
                WindUI:Notify({Title = "防踢", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    tab:Toggle({
        Title = "防甩飞",
        Value = false,
        Callback = function(val)
            local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
            local hum = char:FindFirstChildOfClass("Humanoid")
            local root = char:FindFirstChild("HumanoidRootPart")

            if not hum or not root then
                WindUI:Notify({Title = "错误", Content = "角色未加载完成", Duration = 2, Icon = "x"})
                return
            end

            if val then
                hum.PlatformStand = true
                local antiVelocity = Instance.new("BodyVelocity")
                antiVelocity.Name = "AntiFling"
                antiVelocity.Parent = root
                antiVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                antiVelocity.Velocity = Vector3.new(0, 0, 0)

                local antiAngular = Instance.new("BodyAngularVelocity")
                antiAngular.Name = "AntiFlingAngular"
                antiAngular.Parent = root
                antiAngular.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                antiAngular.AngularVelocity = Vector3.new(0, 0, 0)

                WindUI:Notify({Title = "防甩飞", Content = "已开启", Duration = 2, Icon = "check"})
            else
                hum.PlatformStand = false
                local antiVelocity = root:FindFirstChild("AntiFling")
                if antiVelocity then antiVelocity:Destroy() end
                local antiAngular = root:FindFirstChild("AntiFlingAngular")
                if antiAngular then antiAngular:Destroy() end

                WindUI:Notify({Title = "防甩飞", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    local espEnabled = false
    local espConnections = {}
    local playerAddedConn = nil

    local function createESP(player)
        if not espEnabled or player == LocalPlayer then return end
        
        if player.Character then
            local existing = player.Character:FindFirstChild("NH_ESP")
            if existing then existing:Destroy() end

            local highlight = Instance.new("Highlight")
            highlight.Name = "NH_ESP"
            highlight.FillTransparency = 0.7
            highlight.OutlineTransparency = 0
            highlight.FillColor = Color3.fromRGB(255, 0, 0)
            highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
            highlight.Adornee = player.Character
            highlight.Parent = player.Character
        end

        if not espConnections[player] then
            local connection = player.CharacterAdded:Connect(function(char)
                task.wait(0.5)
                if espEnabled and player ~= LocalPlayer then
                    local existing = char:FindFirstChild("NH_ESP")
                    if existing then existing:Destroy() end

                    local highlight = Instance.new("Highlight")
                    highlight.Name = "NH_ESP"
                    highlight.FillTransparency = 0.7
                    highlight.OutlineTransparency = 0
                    highlight.FillColor = Color3.fromRGB(255, 0, 0)
                    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                    highlight.Adornee = char
                    highlight.Parent = char
                end
            end)
            espConnections[player] = connection
        end
    end

    tab:Toggle({
        Title = "透视玩家",
        Value = false,
        Callback = function(enabled)
            espEnabled = enabled
            if enabled then
                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= LocalPlayer then
                        createESP(player)
                    end
                end

                playerAddedConn = Players.PlayerAdded:Connect(function(player)
                    if espEnabled and player ~= LocalPlayer then
                        createESP(player)
                    end
                end)
            else
                if playerAddedConn then
                    playerAddedConn:Disconnect()
                    playerAddedConn = nil
                end

                for player, conn in pairs(espConnections) do
                    if conn then conn:Disconnect() end
                end
                espConnections = {}

                for _, player in ipairs(Players:GetPlayers()) do
                    if player.Character then
                        local hl = player.Character:FindFirstChild("NH_ESP")
                        if hl then hl:Destroy() end
                    end
                end
            end
            WindUI:Notify({
                Title = "透视玩家",
                Content = enabled and "透视已开启" or "透视已关闭",
                Duration = 2,
                Icon = enabled and "check" or "x"
            })
        end,
    })

    local flyEnabled = false
    local flyConnections = {}
    local flyBody = {}

    tab:Toggle({
        Title = "飞行",
        Value = false,
        Callback = function(val)
            flyEnabled = val
            local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
            local hum = char:FindFirstChildOfClass("Humanoid")
            local root = char:FindFirstChild("HumanoidRootPart")

            if val then
                if hum then
                    for _, state in pairs(Enum.HumanoidStateType:GetEnumItems()) do
                        hum:SetStateEnabled(state, false)
                    end
                    hum:ChangeState(Enum.HumanoidStateType.Swimming)
                end
                if char.Animate then char.Animate.Disabled = true end

                local bg = Instance.new("BodyGyro", root)
                bg.P = 9e4
                bg.maxTorque = Vector3.new(9e9, 9e9, 9e9)
                bg.CFrame = root.CFrame

                local bv = Instance.new("BodyVelocity", root)
                bv.Velocity = Vector3.new(0, 0, 0)
                bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)

                if hum then hum.PlatformStand = true end
                flyBody = {bg = bg, bv = bv}

                local conn = RunService.RenderStepped:Connect(function()
                    if not flyEnabled or not char or not hum or hum.Health <= 0 then
                        if conn then conn:Disconnect() end
                        return
                    end
                    if root then bg.CFrame = workspace.CurrentCamera.CoordinateFrame end
                end)
                table.insert(flyConnections, conn)

                local conn2 = RunService.Heartbeat:Connect(function()
                    if not flyEnabled or not char or not hum then return end
                    if hum.MoveDirection.Magnitude > 0 then
                        char:TranslateBy(hum.MoveDirection * 1)
                    end
                end)
                table.insert(flyConnections, conn2)

                WindUI:Notify({Title = "飞行", Content = "已开启", Duration = 2, Icon = "check"})
            else
                if hum then
                    for _, state in pairs(Enum.HumanoidStateType:GetEnumItems()) do
                        hum:SetStateEnabled(state, true)
                    end
                    hum:ChangeState(Enum.HumanoidStateType.Running)
                    hum.PlatformStand = false
                end
                if char.Animate then char.Animate.Disabled = false end
                if flyBody.bg then flyBody.bg:Destroy() end
                if flyBody.bv then flyBody.bv:Destroy() end
                flyBody = {}

                for _, c in pairs(flyConnections) do
                    pcall(function() c:Disconnect() end)
                end
                flyConnections = {}

                WindUI:Notify({Title = "飞行", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    local function createNoFall()
        local con = nil
        local charCon = nil
        local z = Vector3.zero

        local function bind(c)
            if not c then return end
            local r = c:WaitForChild("HumanoidRootPart", 5)
            if r then
                if con then con:Disconnect() con = nil end
                con = RunService.Heartbeat:Connect(function()
                    if not r.Parent then
                        if con then con:Disconnect() con = nil end
                        return
                    end
                    local v = r.AssemblyLinearVelocity
                    r.AssemblyLinearVelocity = z
                    RunService.RenderStepped:Wait()
                    r.AssemblyLinearVelocity = v
                end)
            end
        end

        local function start()
            if charCon then charCon:Disconnect() end
            bind(LocalPlayer.Character)
            charCon = LocalPlayer.CharacterAdded:Connect(bind)
        end

        local function stop()
            if con then con:Disconnect() con = nil end
            if charCon then charCon:Disconnect() charCon = nil end
        end

        return {start = start, stop = stop}
    end

    tab:Toggle({
        Title = "防摔",
        Value = false,
        Callback = function(val)
            local api = _G.NoFallAPI
            if not api then
                api = createNoFall()
                _G.NoFallAPI = api
            end

            if val then
                api.start()
                WindUI:Notify({Title = "防摔", Content = "已开启", Duration = 2, Icon = "check"})
            else
                api.stop()
                WindUI:Notify({Title = "防摔", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    tab:Toggle({
        Title = "无限跳",
        Value = false,
        Callback = function(val)
            if val then
                local conn = UserInputService.JumpRequest:Connect(function()
                    if LocalPlayer and LocalPlayer.Character then
                        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                        if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
                    end
                end)
                _G.InfJumpCon = conn
                WindUI:Notify({Title = "无限跳", Content = "已开启", Duration = 2, Icon = "check"})
            else
                if _G.InfJumpCon then
                    _G.InfJumpCon:Disconnect()
                    _G.InfJumpCon = nil
                end
                WindUI:Notify({Title = "无限跳", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    local tpConnection = nil
    tab:Toggle({
        Title = "自动存活",
        Value = false,
        Callback = function(val)
            if val then
                if tpConnection then tpConnection:Disconnect() end
                tpConnection = RunService.Stepped:Connect(function()
                    if LocalPlayer and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                        LocalPlayer.Character.HumanoidRootPart.CFrame = CFrame.new(-273, 179.5, 394)
                    end
                end)
                WindUI:Notify({Title = "固定传送", Content = "已开启", Duration = 2, Icon = "check"})
            else
                if tpConnection then
                    tpConnection:Disconnect()
                    tpConnection = nil
                end
                WindUI:Notify({Title = "固定传送", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    tab:Toggle({
        Title = "地图投票",
        Value = false,
        Callback = function(val)
            local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
            local mainGui = playerGui and playerGui:FindFirstChild("MainGui")
            local mapPage = mainGui and mainGui:FindFirstChild("MapVotePage")

            if mapPage then
                mapPage.Visible = val
                WindUI:Notify({
                    Title = "地图投票",
                    Content = val and "界面已开启" or "界面已关闭",
                    Duration = 2,
                    Icon = val and "eye" or "eye-off"
                })
            else
                WindUI:Notify({
                    Title = "提示",
                    Content = "未找到地图投票界面 (MainGui.MapVotePage)",
                    Duration = 2,
                    Icon = "alert"
                })
            end
        end,
    })

    local lockHealthEnabled = false
    local gm = getrawmetatable(game)
    local oldNewIndex = gm.__newindex
    setreadonly(gm, false)
    gm.__newindex = function(t, k, v)
        if lockHealthEnabled and k == "Health" and t:IsA("Humanoid") and t.Parent == LocalPlayer.Character then
            return oldNewIndex(t, k, 100)
        end
        return oldNewIndex(t, k, v)
    end
    setreadonly(gm, true)

    tab:Toggle({
        Title = "锁定血量",
        Value = false,
        Callback = function(val)
            lockHealthEnabled = val
            WindUI:Notify({
                Title = "锁定血量",
                Content = val and "已开启" or "已关闭",
                Duration = 2,
                Icon = val and "check" or "x"
            })
        end,
    })

    local disasterConn = nil
    local disastertable = {
        ["Blizzard"] = "暴风雪",
        ["Sandstorm"] = "沙尘暴",
        ["Tornado"] = "龙卷风",
        ["Volcanic Eruption"] = "火山",
        ["Flash Flood"] = "洪水",
        ["Deadly Virus"] = "病毒",
        ["Tsunami"] = "海啸",
        ["Acid Rain"] = "酸雨",
        ["Fire"] = "火焰",
        ["Meteor Shower"] = "流星雨",
        ["Earthquake"] = "地震",
        ["Thunder Storm"] = "暴风雨",
        ["Avalanche"] = "雪崩",
        ["Lightning"] = "闪电"
    }

    local function checkDisaster(tag)
        if tag and disastertable[tag.Value] then
            WindUI:Notify({
                Title = "灾难预警",
                Content = "⚠️ " .. disastertable[tag.Value] .. " 即将到来！",
                Duration = 5,
                Icon = "alert"
            })
        end
    end

    tab:Toggle({
        Title = "灾难预警",
        Value = false,
        Callback = function(val)
            if val then
                local SurvivalTag = workspace:FindFirstChild("SurvivalTag")
                if SurvivalTag then
                    checkDisaster(SurvivalTag)
                end
                if disasterConn then disasterConn:Disconnect() end
                disasterConn = workspace.ChildAdded:Connect(function(v)
                    if v.Name == "SurvivalTag" then
                        checkDisaster(v)
                    end
                end)
                WindUI:Notify({Title = "灾难预警", Content = "已开启", Duration = 2, Icon = "check"})
            else
                if disasterConn then
                    disasterConn:Disconnect()
                    disasterConn = nil
                end
                WindUI:Notify({Title = "灾难预警", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    local voidConnection = nil
    local savedVoidCFrame = nil
    local VOID_POSITION = CFrame.new(0, -10000, 0)

    tab:Toggle({
        Title = "虚空传送",
        Value = false,
        Callback = function(val)
            local char = LocalPlayer.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")

            if val then
                if root then
                    savedVoidCFrame = root.CFrame
                    if voidConnection then voidConnection:Disconnect() end
                    voidConnection = RunService.Heartbeat:Connect(function()
                        local cChar = LocalPlayer.Character
                        local cRoot = cChar and cChar:FindFirstChild("HumanoidRootPart")
                        if cRoot then
                            cRoot.CFrame = VOID_POSITION
                            cRoot.AssemblyLinearVelocity = Vector3.zero
                            cRoot.AssemblyAngularVelocity = Vector3.zero
                            for _, child in ipairs(cRoot:GetChildren()) do
                                if child:IsA("BodyVelocity") or child:IsA("BodyThrust") or child:IsA("BodyAngularVelocity") or child:IsA("BodyGyro") then
                                    child:Destroy()
                                end
                            end
                        end
                    end)
                    WindUI:Notify({Title = "虚空传送", Content = "已进入虚空", Duration = 2, Icon = "check"})
                end
            else
                if voidConnection then
                    voidConnection:Disconnect()
                    voidConnection = nil
                end
                if savedVoidCFrame and root then
                    root.CFrame = savedVoidCFrame
                end
                WindUI:Notify({Title = "虚空传送", Content = "已返回现实", Duration = 2, Icon = "x"})
            end
        end,
    })

    local function playActionSound()
        local sound = Instance.new("Sound", LocalPlayer:WaitForChild("PlayerGui"))
        sound.SoundId = "rbxassetid://942127495"
        sound.Volume = 1
        sound:Play()
        game.Debris:AddItem(sound, 2)
    end

    local function setTransparency(character, transparency)
        for _, part in pairs(character:GetDescendants()) do
            if part:IsA("BasePart") or part:IsA("Decal") then
                part.Transparency = transparency
            end
        end
    end

    local invis_on = false
    tab:Toggle({
        Title = "隐身",
        Value = false,
        Callback = function(val)
            invis_on = val
            playActionSound()

            if invis_on then
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    local savedpos = char.HumanoidRootPart.CFrame
                    task.wait()
                    char:MoveTo(Vector3.new(-25.95, 84, 3537.55))
                    task.wait(0.15)
                    
                    local Seat = Instance.new('Seat', workspace)
                    Seat.Anchored = false
                    Seat.CanCollide = false
                    Seat.Name = 'invischair'
                    Seat.Transparency = 1
                    Seat.Position = Vector3.new(-25.95, 84, 3537.55)
                    
                    local Weld = Instance.new("Weld", Seat)
                    Weld.Part0 = Seat
                    Weld.Part1 = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
                    
                    task.wait()
                    Seat.CFrame = savedpos
                    setTransparency(char, 0.5)
                    WindUI:Notify({Title = "隐身", Content = "已开启", Duration = 2, Icon = "check"})
                end
            else
                local invisChair = workspace:FindFirstChild('invischair')
                if invisChair then
                    invisChair:Destroy()
                end
                if LocalPlayer.Character then
                    setTransparency(LocalPlayer.Character, 0)
                end
                WindUI:Notify({Title = "隐身", Content = "已关闭", Duration = 2, Icon = "x"})
            end
        end,
    })

    local defaultSpeed = 16
    local boostedSpeed = 48
    local isSpeedBoosted = false
    local speedCon = nil

    tab:Toggle({
        Title = "加速",
        Value = false,
        Callback = function(val)
            isSpeedBoosted = val
            playActionSound()
            local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
            
            if isSpeedBoosted then
                if humanoid then
                    humanoid.WalkSpeed = boostedSpeed
                end
                if not speedCon then
                    speedCon = LocalPlayer.CharacterAdded:Connect(function(character)
                        local hum = character:WaitForChild("Humanoid")
                        hum.WalkSpeed = isSpeedBoosted and boostedSpeed or defaultSpeed
                    end)
                end
                WindUI:Notify({Title = "加速", Content = "速度提升至: " .. boostedSpeed, Duration = 2, Icon = "check"})
            else
                if humanoid then
                    humanoid.WalkSpeed = defaultSpeed
                end
                if speedCon then
                    speedCon:Disconnect()
                    speedCon = nil
                end
                WindUI:Notify({Title = "加速", Content = "速度恢复至: " .. defaultSpeed, Duration = 2, Icon = "x"})
            end
        end,
    })

    return tab
end
