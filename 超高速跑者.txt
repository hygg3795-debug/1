return function(Window, WindUI)
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local player = Players.LocalPlayer

    local tab = Window:Tab({Title = "超高速跑者", Icon = " "})

    local TARGET_POSITION = CFrame.new(0, 6.5, -9076)
    local baseY = 1
    local minOffset = baseY - 3
    local maxOffset = baseY + 4
    local currentY = baseY
    local speed = 0.5
    local dir = 1

    local running = false
    local loopConnection = nil

    tab:Toggle({
        Title = "自动胜利",
        Value = false,
        Callback = function(val)
            running = val
            if running then
                currentY = baseY
                dir = 1
                loopConnection = RunService.Heartbeat:Connect(function()
                    local character = player.Character
                    if not character then return end
                    local hrp = character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        currentY = currentY + speed * dir
                        if currentY <= minOffset then
                            dir = 1
                            currentY = minOffset
                        end
                        if currentY >= maxOffset then
                            dir = -1
                            currentY = maxOffset
                        end
                        hrp.CFrame = TARGET_POSITION + Vector3.new(0, currentY, 0)
                    end
                end)
            else
                if loopConnection then
                    loopConnection:Disconnect()
                    loopConnection = nil
                end
            end
        end
    })

    return tab
end
