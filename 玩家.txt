return function(Window, WindUI)
    local tab = Window:Tab({Title = "玩家", Icon = "users"})

    local flingSection = tab:Section({ Title = "甩飞控制", Box = true })

    local LocalPlayer = game.Players.LocalPlayer
    local Module = {
        state = {
            isLoopFlingingSingle = false,
            isLoopFlingingAll = false,
            targetPlayer = nil
        }
    }

    local function SkidFling(TargetPlayer)
        if not TargetPlayer or TargetPlayer == LocalPlayer then return end
        local Character = LocalPlayer.Character
        local RootPart = Character and Character:FindFirstChild("HumanoidRootPart")
        local TCharacter = TargetPlayer.Character
        local TRootPart = TCharacter and TCharacter:FindFirstChild("HumanoidRootPart")

        if not (Character and RootPart and TCharacter and TRootPart) then return end

        local SavedCFrame = RootPart.CFrame
        local OldVelocity = RootPart.AssemblyLinearVelocity

        local bav = Instance.new("BodyAngularVelocity")
        bav.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        bav.AngularVelocity = Vector3.new(0, 999999, 0)
        bav.Parent = RootPart

        local startTime = tick()
        while tick() - startTime < 0.5 do
            if not TargetPlayer or not TargetPlayer.Parent or not TRootPart or not TRootPart.Parent then break end
            RootPart.CFrame = TRootPart.CFrame * CFrame.new(0, 0, 0)
            RootPart.AssemblyLinearVelocity = Vector3.new(9999, 9999, 9999)
            task.wait()
        end

        bav:Destroy()
        RootPart.AssemblyLinearVelocity = OldVelocity
        RootPart.CFrame = SavedCFrame
    end

    local function startFlingAll()
        local players = game.Players:GetPlayers()
        for _, player in ipairs(players) do
            if player ~= LocalPlayer then
                task.spawn(function()
                    SkidFling(player)
                end)
            end
        end
        WindUI:Notify({Title = "成功", Content = "甩飞所有玩家一次", Duration = 3, Icon = "check"})
    end

    local function stopLoopFlingSingle()
        Module.state.isLoopFlingingSingle = false
        WindUI:Notify({Title = "停止", Content = "循环甩飞玩家已停止", Duration = 3, Icon = "x"})
    end

    local function stopLoopFlingAll()
        Module.state.isLoopFlingingAll = false
        WindUI:Notify({Title = "停止", Content = "循环甩飞所有玩家已停止", Duration = 3, Icon = "x"})
    end

    -- 后台循环处理
    task.spawn(function()
        while true do
            task.wait(0.6)
            if Module.state.isLoopFlingingAll then
                local players = game.Players:GetPlayers()
                for _, player in ipairs(players) do
                    if not Module.state.isLoopFlingingAll then break end
                    if player ~= LocalPlayer then
                        SkidFling(player)
                    end
                end
            elseif Module.state.isLoopFlingingSingle and Module.state.targetPlayer then
                SkidFling(Module.state.targetPlayer)
            end
        end
    end)

    -- 玩家选择下拉框
    local playerMap = {}
    local function getPlayerNames()
        local names = {}
        playerMap = {}
        for _, p in ipairs(game.Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local label = p.DisplayName .. " (@" .. p.Name .. ")"
                table.insert(names, label)
                playerMap[label] = p
            end
        end
        return names
    end

    flingSection:Dropdown({
        Title = "选择目标玩家",
        Values = getPlayerNames(),
        Callback = function(selected)
            Module.state.targetPlayer = playerMap[selected]
        end,
    })

    flingSection:Button({
        Title = "甩飞选中玩家 (一次)",
        Callback = function()
            if Module.state.targetPlayer then
                task.spawn(function()
                    SkidFling(Module.state.targetPlayer)
                end)
                WindUI:Notify({Title = "成功", Content = "已甩飞目标玩家", Duration = 3, Icon = "check"})
            else
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 3, Icon = "alert"})
            end
        end,
    })

    flingSection:Toggle({
        Title = "循环甩飞选中玩家",
        Value = false,
        Callback = function(val)
            if val then
                if not Module.state.targetPlayer then
                    WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 3, Icon = "alert"})
                    return
                end
                Module.state.isLoopFlingingSingle = true
                WindUI:Notify({Title = "开启", Content = "已开启循环甩飞目标", Duration = 3, Icon = "check"})
            else
                stopLoopFlingSingle()
            end
        end,
    })

    flingSection:Button({
        Title = "甩飞所有玩家 (一次)",
        Callback = function()
            startFlingAll()
        end,
    })

    flingSection:Toggle({
        Title = "循环甩飞所有人",
        Value = false,
        Callback = function(val)
            if val then
                Module.state.isLoopFlingingAll = true
                WindUI:Notify({Title = "开启", Content = "循环甩飞所有玩家已开启", Duration = 3, Icon = "check"})
            else
                stopLoopFlingAll()
            end
        end,
    })

    return tab
end
