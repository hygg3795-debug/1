return function(Window, WindUI)
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local TweenService = game:GetService("TweenService")
    local Workspace = game:GetService("Workspace")
    local LocalPlayer = Players.LocalPlayer

    local tab = Window:Tab({Title = "玩家", Icon = "users"})
    local targetPlayer = nil
    local playerMap = {}

    local function getPlayerNames()
        local names = {}
        playerMap = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local label = p.DisplayName .. " (@" .. p.Name .. ")"
                table.insert(names, label)
                playerMap[label] = p
            end
        end
        return names
    end

    local playerSection = tab:Section({Title = "玩家选项", Box = true})
    local playerDropdown = playerSection:Dropdown({
        Title = "选择目标玩家",
        Values = getPlayerNames(),
        Callback = function(selected)
            if selected then
                targetPlayer = playerMap[selected]
            end
        end
    })

    local function refreshDropdown()
        local newNames = getPlayerNames()
        if playerDropdown then
            if playerDropdown.SetValues then
                playerDropdown:SetValues(newNames)
            elseif playerDropdown.Update then
                playerDropdown:Update(newNames)
            elseif playerDropdown.UpdateValues then
                playerDropdown:UpdateValues(newNames)
            elseif playerDropdown.Refresh then
                playerDropdown:Refresh(newNames)
            else
                local oldCallback = playerDropdown.Callback
                if playerDropdown.Remove then
                    playerDropdown:Remove()
                elseif playerDropdown.Destroy then
                    playerDropdown:Destroy()
                end
                playerDropdown = playerSection:Dropdown({
                    Title = "选择目标玩家",
                    Values = newNames,
                    Callback = oldCallback or function(selected)
                        if selected then
                            targetPlayer = playerMap[selected]
                        end
                    end
                })
            end
        else
            playerDropdown = playerSection:Dropdown({
                Title = "选择目标玩家",
                Values = newNames,
                Callback = function(selected)
                    if selected then
                        targetPlayer = playerMap[selected]
                    end
                end
            })
        end
    end

    playerSection:Button({Title = "刷新玩家列表", Callback = function()
        refreshDropdown()
        WindUI:Notify({Title = "刷新成功", Content = "玩家列表已更新", Duration = 2, Icon = "check"})
    end})

    Players.PlayerAdded:Connect(refreshDropdown)
    Players.PlayerRemoving:Connect(refreshDropdown)

    local teleportSection = playerSection:Section({Title = "传送", Collapsible = true, Box = true})

    teleportSection:Toggle({Title = "平滑跟随", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            _G.SmoothFollowCon = RunService.Heartbeat:Connect(function()
                local myChar = LocalPlayer.Character
                if not myChar then return end
                local myRoot = myChar:FindFirstChild("HumanoidRootPart")
                if not myRoot then return end
                local target = targetPlayer
                if not target or not target.Character then return end
                local targetRoot = target.Character:FindFirstChild("HumanoidRootPart")
                if not targetRoot then return end
                local offset = targetRoot.CFrame.LookVector * -2.5
                myRoot.CFrame = CFrame.new(targetRoot.Position + offset, targetRoot.Position)
            end)
        else
            if _G.SmoothFollowCon then
                _G.SmoothFollowCon:Disconnect()
                _G.SmoothFollowCon = nil
            end
        end
    end})

    teleportSection:Button({Title = "传送至选中玩家", Callback = function()
        if not targetPlayer then
            WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
            return
        end
        local target = targetPlayer
        if not target.Character or not target.Character:FindFirstChild("HumanoidRootPart") then return end
        local myChar = LocalPlayer.Character
        if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return end
        local targetPos = target.Character.HumanoidRootPart.Position
        myChar.HumanoidRootPart.CFrame = CFrame.new(targetPos + Vector3.new(0, 0, 3))
    end})

    teleportSection:Button({Title = "传送最近玩家", Callback = function()
        local nearest = nil
        local minDist = math.huge
        local myChar = LocalPlayer.Character
        if not myChar then return end
        local myRoot = myChar:FindFirstChild("HumanoidRootPart")
        if not myRoot then return end
        local myPos = myRoot.Position
        
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local pChar = p.Character
                if pChar then
                    local pRoot = pChar:FindFirstChild("HumanoidRootPart")
                    if pRoot then
                        local dist = (pRoot.Position - myPos).Magnitude
                        if dist < minDist then
                            minDist = dist
                            nearest = p
                        end
                    end
                end
            end
        end
        
        if nearest then
            local targetRoot = nearest.Character and nearest.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot and myRoot then
                myRoot.CFrame = targetRoot.CFrame + Vector3.new(0, 0, 3)
                WindUI:Notify({Title = "传送成功", Content = "已传送至 " .. nearest.DisplayName, Duration = 2, Icon = "check"})
            end
        else
            WindUI:Notify({Title = "提示", Content = "未找到可传送的玩家", Duration = 2, Icon = "alert"})
        end
    end})

    local teleportAllConnection = nil
    local teleportAllRunning = false
    teleportSection:Toggle({Title = "传送所有人 (0.45s间隔)", Value = false, Callback = function(val)
        teleportAllRunning = val
        if val then
            local players = Players:GetPlayers()
            local count = 0
            for _, p in ipairs(players) do
                if not teleportAllRunning then break end
                if p ~= LocalPlayer then
                    local char = p.Character
                    if char and char:FindFirstChild("HumanoidRootPart") then
                        local myChar = LocalPlayer.Character
                        if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                            myChar.HumanoidRootPart.CFrame = char.HumanoidRootPart.CFrame + Vector3.new(0, 0, 3)
                            count = count + 1
                            task.wait(0.45)
                        end
                    end
                end
            end
            if count > 0 then
                WindUI:Notify({Title = "传送所有人", Content = "已传送至 " .. count .. " 名玩家", Duration = 2, Icon = "check"})
            else
                WindUI:Notify({Title = "传送所有人", Content = "没有可传送的玩家", Duration = 2, Icon = "alert"})
            end
        end
    end})

    playerSection:Divider()

    local trollingSection = playerSection:Section({Title = "恶搞", Collapsible = true, Box = true})

    local suckConnection = nil
    local suckAnimation = nil
    local originalGravity = nil
    trollingSection:Toggle({Title = "口人", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            local target = targetPlayer
            local myChar = LocalPlayer.Character
            if not myChar then return end
            local myRoot = myChar:FindFirstChild("HumanoidRootPart")
            local targetChar = target.Character
            if not targetChar then return end
            local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
            if not myRoot or not targetRoot then return end
            originalGravity = Workspace.Gravity
            Workspace.Gravity = 0
            local animation = Instance.new("Animation")
            animation.AnimationId = "rbxassetid://5918726674"
            local humanoid = myChar:FindFirstChildOfClass("Humanoid")
            if humanoid then
                suckAnimation = humanoid:LoadAnimation(animation)
                suckAnimation:Play()
                suckAnimation:AdjustSpeed(1)
            end
            local targetTorso = targetChar:FindFirstChild("LowerTorso") or targetChar:FindFirstChild("UpperTorso")
            if targetTorso then
                suckConnection = RunService.Heartbeat:Connect(function()
                    if myRoot and targetTorso then
                        myRoot.CFrame = targetTorso.CFrame * CFrame.new(0, -2.3, -1.0) * CFrame.Angles(0, math.pi, 0)
                    end
                end)
            end
        else
            if suckConnection then suckConnection:Disconnect() suckConnection = nil end
            if suckAnimation then suckAnimation:Stop() suckAnimation = nil end
            if originalGravity then Workspace.Gravity = originalGravity originalGravity = nil end
        end
    end})

    local bangConnection = nil
    local bangAnimation = nil
    trollingSection:Toggle({Title = "配人", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            local target = targetPlayer
            local myChar = LocalPlayer.Character
            if not myChar then return end
            local myRoot = myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then return end
            local animation = Instance.new("Animation")
            animation.AnimationId = "rbxassetid://10714068222"
            local humanoid = myChar:FindFirstChildOfClass("Humanoid")
            if humanoid then
                bangAnimation = humanoid:LoadAnimation(animation)
                bangAnimation:Play()
                bangAnimation:AdjustSpeed(2)
            end
            bangConnection = RunService.Heartbeat:Connect(function()
                if not val then return end
                local targetChar = target.Character
                if not targetChar then return end
                local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
                if not targetHRP or not myRoot then return end
                local forwardCFrame = targetHRP.CFrame * CFrame.new(0, 0, 1)
                local backwardCFrame = targetHRP.CFrame * CFrame.new(0, 0, 2.5)
                local tweenForward = TweenService:Create(myRoot, TweenInfo.new(0.2, Enum.EasingStyle.Linear, Enum.EasingDirection.Out), {CFrame = forwardCFrame})
                tweenForward:Play()
                tweenForward.Completed:Wait()
                local tweenBackward = TweenService:Create(myRoot, TweenInfo.new(0.2, Enum.EasingStyle.Linear, Enum.EasingDirection.Out), {CFrame = backwardCFrame})
                tweenBackward:Play()
                tweenBackward.Completed:Wait()
            end)
        else
            if bangConnection then bangConnection:Disconnect() bangConnection = nil end
            if bangAnimation then bangAnimation:Stop() bangAnimation = nil end
        end
    end})

    local susConnection = nil
    local susAnimation = nil
    trollingSection:Toggle({Title = "被配", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            local target = targetPlayer
            local myChar = LocalPlayer.Character
            if not myChar then return end
            local myRoot = myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then return end
            local animation = Instance.new("Animation")
            animation.AnimationId = "rbxassetid://10714360343"
            local humanoid = myChar:FindFirstChildOfClass("Humanoid")
            if humanoid then
                susAnimation = humanoid:LoadAnimation(animation)
                susAnimation:Play()
            end
            susConnection = RunService.Heartbeat:Connect(function()
                if not val then return end
                local targetChar = target.Character
                if not targetChar then return end
                local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
                if not targetHRP or not myRoot then return end
                local forwardCFrame = targetHRP.CFrame * CFrame.new(0, 0, -1.5)
                local backwardCFrame = targetHRP.CFrame * CFrame.new(0, 0, -1.1)
                local tweenForward = TweenService:Create(myRoot, TweenInfo.new(0.15, Enum.EasingStyle.Linear, Enum.EasingDirection.Out), {CFrame = forwardCFrame})
                tweenForward:Play()
                tweenForward.Completed:Wait()
                local tweenBackward = TweenService:Create(myRoot, TweenInfo.new(0.15, Enum.EasingStyle.Linear, Enum.EasingDirection.Out), {CFrame = backwardCFrame})
                tweenBackward:Play()
                tweenBackward.Completed:Wait()
            end)
        else
            if susConnection then susConnection:Disconnect() susConnection = nil end
            if susAnimation then susAnimation:Stop() susAnimation = nil end
        end
    end})

    local getSuckedConnection = nil
    local getSuckedAnimation = nil
    local getSuckedGravity = nil
    local getSuckedRunning = false
    trollingSection:Toggle({Title = "人口", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            getSuckedRunning = true
            local target = targetPlayer
            local myChar = LocalPlayer.Character
            if not myChar then return end
            local myRoot = myChar:FindFirstChild("HumanoidRootPart")
            local targetChar = target.Character
            if not targetChar then return end
            local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
            if not myRoot or not targetRoot then return end
            getSuckedGravity = Workspace.Gravity
            Workspace.Gravity = 0
            coroutine.wrap(function()
                while getSuckedRunning and myRoot and targetRoot and myRoot.Position.Y <= 44 do
                    task.wait()
                    myRoot.CFrame = myRoot.CFrame * CFrame.new(0, 1.5, 0)
                end
                task.wait(1)
                if getSuckedRunning then
                    local animation = Instance.new("Animation")
                    animation.AnimationId = "rbxassetid://5918726674"
                    local humanoid = myChar:FindFirstChildOfClass("Humanoid")
                    if humanoid then
                        getSuckedAnimation = humanoid:LoadAnimation(animation)
                        getSuckedAnimation:Play()
                        getSuckedAnimation:AdjustSpeed(1)
                    end
                    getSuckedConnection = RunService.Stepped:Connect(function()
                        if getSuckedRunning and myRoot and targetRoot then
                            myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 2.3, -1.1) * CFrame.Angles(0, math.pi, 0)
                            myRoot.Velocity = Vector3.new(0, 0, 0)
                        end
                    end)
                end
            end)()
        else
            getSuckedRunning = false
            if getSuckedConnection then getSuckedConnection:Disconnect() getSuckedConnection = nil end
            if getSuckedAnimation then getSuckedAnimation:Stop() getSuckedAnimation = nil end
            if getSuckedGravity then Workspace.Gravity = getSuckedGravity getSuckedGravity = nil end
        end
    end})

    local cameraFollowConnection = nil
    local isCameraFollowing = false
    trollingSection:Toggle({Title = "镜头跟随", Value = false, Callback = function(val)
        isCameraFollowing = val
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            local target = targetPlayer
            local targetChar = target.Character
            if not targetChar then
                WindUI:Notify({Title = "提示", Content = "目标玩家没有角色", Duration = 2, Icon = "alert"})
                return
            end
            local targetHumanoid = targetChar:FindFirstChildOfClass("Humanoid")
            if not targetHumanoid then
                WindUI:Notify({Title = "提示", Content = "目标玩家没有Humanoid", Duration = 2, Icon = "alert"})
                return
            end
            workspace.CurrentCamera.CameraSubject = targetHumanoid
            cameraFollowConnection = RunService.Heartbeat:Connect(function()
                if not isCameraFollowing then return end
                local currentTarget = targetPlayer
                if not currentTarget or not currentTarget.Character then return end
                local currentHumanoid = currentTarget.Character:FindFirstChildOfClass("Humanoid")
                if currentHumanoid then
                    workspace.CurrentCamera.CameraSubject = currentHumanoid
                end
            end)
            WindUI:Notify({Title = "镜头跟随", Content = "已锁定 " .. target.DisplayName, Duration = 2, Icon = "check"})
        else
            if cameraFollowConnection then
                cameraFollowConnection:Disconnect()
                cameraFollowConnection = nil
            end
            local myChar = LocalPlayer.Character
            if myChar then
                local myHumanoid = myChar:FindFirstChildOfClass("Humanoid")
                if myHumanoid then
                    workspace.CurrentCamera.CameraSubject = myHumanoid
                end
            end
            WindUI:Notify({Title = "镜头跟随", Content = "已关闭", Duration = 2, Icon = "check"})
        end
    end})

    playerSection:Divider()

    local flingSection = playerSection:Section({Title = "甩飞", Collapsible = true, Box = true})

    local function skidFling(targetPlayer)
        if not targetPlayer then return end
        local targetChar = targetPlayer.Character
        if not targetChar then return end

        local myChar = LocalPlayer.Character
        if not myChar then return end

        local myHumanoid = myChar:FindFirstChildOfClass("Humanoid")
        local myRoot = myHumanoid and myHumanoid.RootPart
        if not myRoot then return end

        local targetHumanoid = targetChar:FindFirstChildOfClass("Humanoid")
        local targetRoot = targetHumanoid and targetHumanoid.RootPart
        local targetHead = targetChar:FindFirstChild("Head")

        local oldPos = myRoot.CFrame
        local oldVel = myRoot.AssemblyLinearVelocity
        local oldRotVel = myRoot.AssemblyAngularVelocity

        local myHumanoidObject = myChar:FindFirstChildOfClass("Humanoid")
        local originalSubject = workspace.CurrentCamera.CameraSubject
        if myHumanoidObject then
            workspace.CurrentCamera.CameraSubject = myHumanoidObject
        end

        local function forceMove(basePart, posOffset, angOffset)
            if not basePart or not basePart.Parent then return end
            myRoot.CFrame = CFrame.new(basePart.Position) * posOffset * angOffset
            myChar:SetPrimaryPartCFrame(CFrame.new(basePart.Position) * posOffset * angOffset)
            myRoot.AssemblyLinearVelocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            myRoot.AssemblyAngularVelocity = Vector3.new(9e8, 9e8, 9e8)
        end

        local function flingLoop(basePart)
            local timeLimit = 2
            local startTime = tick()
            local angle = 0

            repeat
                if not myRoot or not myRoot.Parent or not targetHumanoid then break end

                if basePart.AssemblyLinearVelocity.Magnitude < 50 then
                    angle = angle + 100
                    forceMove(basePart, CFrame.new(0, 1.5, 0) + targetHumanoid.MoveDirection * basePart.AssemblyLinearVelocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, 0) + targetHumanoid.MoveDirection * basePart.AssemblyLinearVelocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(2.25, 1.5, -2.25) + targetHumanoid.MoveDirection * basePart.AssemblyLinearVelocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(-2.25, -1.5, 2.25) + targetHumanoid.MoveDirection * basePart.AssemblyLinearVelocity.Magnitude / 1.25, CFrame.Angles(math.rad(angle), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, 1.5, 0) + targetHumanoid.MoveDirection, CFrame.Angles(math.rad(angle), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, 0) + targetHumanoid.MoveDirection, CFrame.Angles(math.rad(angle), 0, 0))
                    task.wait()
                else
                    forceMove(basePart, CFrame.new(0, 1.5, targetHumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, -targetHumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, 1.5, targetHumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, 1.5, targetRoot.AssemblyLinearVelocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, -targetRoot.AssemblyLinearVelocity.Magnitude / 1.25), CFrame.Angles(0, 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, 1.5, targetRoot.AssemblyLinearVelocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(-90), 0, 0))
                    task.wait()
                    forceMove(basePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                    task.wait()
                end
            until basePart.AssemblyLinearVelocity.Magnitude > 500
                or not targetChar.Parent
                or targetHumanoid.Sit
                or myHumanoid.Health <= 0
                or tick() > startTime + timeLimit
        end

        local targetPart = nil
        if targetRoot and targetHead then
            if (targetRoot.Position - targetHead.Position).Magnitude > 5 then
                targetPart = targetHead
            else
                targetPart = targetRoot
            end
        elseif targetRoot then
            targetPart = targetRoot
        elseif targetHead then
            targetPart = targetHead
        else
            WindUI:Notify({Title = "甩飞失败", Content = "目标没有可撞击的身体部件", Duration = 2, Icon = "alert"})
            return
        end

        myHumanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        local bv = Instance.new("BodyVelocity")
        bv.Name = "FlingVelocity"
        bv.Parent = myRoot
        bv.Velocity = Vector3.new(9e8, 9e8, 9e8)
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)

        flingLoop(targetPart)

        bv:Destroy()

        myHumanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)

        myRoot.CFrame = oldPos
        myRoot.AssemblyLinearVelocity = oldVel
        myRoot.AssemblyAngularVelocity = oldRotVel

        if originalSubject then
            workspace.CurrentCamera.CameraSubject = originalSubject
        elseif myHumanoidObject then
            workspace.CurrentCamera.CameraSubject = myHumanoidObject
        end

        task.wait(0.05)
        myHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
    end

    flingSection:Button({Title = "甩飞当前选中", Callback = function()
        if not targetPlayer then
            WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
            return
        end
        local target = targetPlayer
        if not target or not target.Character then
            WindUI:Notify({Title = "提示", Content = "目标玩家不在游戏中", Duration = 2, Icon = "alert"})
            return
        end
        local targetHum = target.Character:FindFirstChildOfClass("Humanoid")
        if not targetHum or targetHum.Health <= 0 then
            WindUI:Notify({Title = "提示", Content = "目标已死亡", Duration = 2, Icon = "alert"})
            return
        end
        skidFling(target)
        WindUI:Notify({Title = "甩飞成功", Content = "已弹飞 " .. target.DisplayName, Duration = 1.5, Icon = "check"})
    end})

    local loopFlingRunning = false
    local loopFlingConnection = nil
    flingSection:Toggle({Title = "循环甩飞当前选中", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            loopFlingRunning = true
            local target = targetPlayer
            WindUI:Notify({Title = "循环甩飞", Content = "已开启，目标: " .. target.DisplayName, Duration = 2, Icon = "check"})

            loopFlingConnection = RunService.Heartbeat:Connect(function()
                if not loopFlingRunning then
                    loopFlingConnection:Disconnect()
                    loopFlingConnection = nil
                    return
                end
                if not targetPlayer or not targetPlayer.Character then
                    return
                end
                local targetHum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
                if not targetHum or targetHum.Health <= 0 then
                    return
                end
                skidFling(targetPlayer)
                task.wait(0.3)
            end)
        else
            loopFlingRunning = false
            if loopFlingConnection then
                loopFlingConnection:Disconnect()
                loopFlingConnection = nil
            end
            WindUI:Notify({Title = "循环甩飞", Content = "已关闭", Duration = 1.5, Icon = "check"})
        end
    end})

    flingSection:Button({Title = "甩飞所有玩家", Callback = function()
        local count = 0
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local char = p.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        skidFling(p)
                        count = count + 1
                        task.wait(0.2)
                    end
                end
            end
        end
        WindUI:Notify({Title = "甩飞所有玩家", Content = "已甩飞 " .. count .. " 名玩家", Duration = 2, Icon = "check"})
    end})

    local loopAllFlingRunning = false
    local loopAllFlingConnection = nil
    flingSection:Toggle({Title = "循环甩飞所有玩家", Value = false, Callback = function(val)
        if val then
            loopAllFlingRunning = true
            WindUI:Notify({Title = "循环甩飞所有玩家", Content = "已开启", Duration = 2, Icon = "check"})

            loopAllFlingConnection = RunService.Heartbeat:Connect(function()
                if not loopAllFlingRunning then
                    loopAllFlingConnection:Disconnect()
                    loopAllFlingConnection = nil
                    return
                end
                local players = Players:GetPlayers()
                for _, p in ipairs(players) do
                    if not loopAllFlingRunning then break end
                    if p ~= LocalPlayer then
                        local char = p.Character
                        if char then
                            local hum = char:FindFirstChildOfClass("Humanoid")
                            if hum and hum.Health > 0 then
                                skidFling(p)
                                task.wait(0.2)
                            end
                        end
                    end
                end
                task.wait(0.3)
            end)
        else
            loopAllFlingRunning = false
            if loopAllFlingConnection then
                loopAllFlingConnection:Disconnect()
                loopAllFlingConnection = nil
            end
            WindUI:Notify({Title = "循环甩飞所有玩家", Content = "已关闭", Duration = 1.5, Icon = "check"})
        end
    end})

    playerSection:Divider()

    local fling2Section = playerSection:Section({Title = "甩飞2.0", Collapsible = true, Box = true})

    local function teleportFling(targetPlayer)
        if not targetPlayer then return end
        local targetChar = targetPlayer.Character
        if not targetChar then return end
        local targetHrp = targetChar:FindFirstChild("HumanoidRootPart")
        if not targetHrp then return end
        
        local myChar = LocalPlayer.Character
        if not myChar then return end
        local myHrp = myChar:FindFirstChild("HumanoidRootPart")
        if not myHrp then return end
        
        local originalPos = myHrp.Position
        local startTime = tick()
        local direction = 1
        local lastSwitch = tick()
        
        while tick() - startTime < 3 do
            local currentTargetChar = targetPlayer.Character
            if not currentTargetChar then break end
            local currentTargetHrp = currentTargetChar:FindFirstChild("HumanoidRootPart")
            if not currentTargetHrp then break end
            
            if tick() - lastSwitch > 0.15 then
                direction = direction * -1
                lastSwitch = tick()
            end
            
            local myChar2 = LocalPlayer.Character
            local myHrp2 = myChar2 and myChar2:FindFirstChild("HumanoidRootPart")
            if myHrp2 then
                local offset = direction * 2
                local targetPos = currentTargetHrp.Position + currentTargetHrp.CFrame.LookVector * offset
                myHrp2.CFrame = CFrame.new(targetPos)
                myHrp2.AssemblyAngularVelocity = Vector3.new(50000, 50000, 50000)
                myHrp2.AssemblyLinearVelocity = Vector3.new(9e7, 9e7, 9e7)
            end
            task.wait(0.05)
        end
        
        local myChar3 = LocalPlayer.Character
        local myHrp3 = myChar3 and myChar3:FindFirstChild("HumanoidRootPart")
        if myHrp3 and originalPos then
            myHrp3.Position = originalPos
            myHrp3.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            myHrp3.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end
    end

    fling2Section:Button({Title = "甩飞当前选中", Callback = function()
        if not targetPlayer then
            WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
            return
        end
        local target = targetPlayer
        if not target or not target.Character then
            WindUI:Notify({Title = "提示", Content = "目标玩家不在游戏中", Duration = 2, Icon = "alert"})
            return
        end
        local targetHum = target.Character:FindFirstChildOfClass("Humanoid")
        if not targetHum or targetHum.Health <= 0 then
            WindUI:Notify({Title = "提示", Content = "目标已死亡", Duration = 2, Icon = "alert"})
            return
        end
        teleportFling(target)
        WindUI:Notify({Title = "甩飞成功", Content = "已弹飞 " .. target.DisplayName, Duration = 1.5, Icon = "check"})
    end})

    local loopFling2Running = false
    local loopFling2Connection = nil
    local loopFling2Target = nil
    fling2Section:Toggle({Title = "循环甩飞当前选中", Value = false, Callback = function(val)
        if val then
            if not targetPlayer then
                WindUI:Notify({Title = "提示", Content = "请先选择目标玩家", Duration = 2, Icon = "alert"})
                return
            end
            loopFling2Running = true
            loopFling2Target = targetPlayer
            WindUI:Notify({Title = "循环甩飞", Content = "已开启，目标: " .. loopFling2Target.DisplayName, Duration = 2, Icon = "check"})

            loopFling2Connection = RunService.Heartbeat:Connect(function()
                if not loopFling2Running then
                    loopFling2Connection:Disconnect()
                    loopFling2Connection = nil
                    return
                end
                if not loopFling2Target or not loopFling2Target.Character then
                    return
                end
                local targetHum = loopFling2Target.Character:FindFirstChildOfClass("Humanoid")
                if not targetHum or targetHum.Health <= 0 then
                    return
                end
                local targetChar = loopFling2Target.Character
                local targetHrp = targetChar:FindFirstChild("HumanoidRootPart")
                if not targetHrp then return end
                
                local myChar = LocalPlayer.Character
                if not myChar then return end
                local myHrp = myChar:FindFirstChild("HumanoidRootPart")
                if not myHrp then return end
                
                local direction = math.sin(tick() * 6)
                local offset = direction * 2.5
                local targetPos = targetHrp.Position + targetHrp.CFrame.LookVector * offset
                myHrp.CFrame = CFrame.new(targetPos)
                myHrp.AssemblyAngularVelocity = Vector3.new(50000, 50000, 50000)
                myHrp.AssemblyLinearVelocity = Vector3.new(9e7, 9e7, 9e7)
            end)
        else
            loopFling2Running = false
            if loopFling2Connection then
                loopFling2Connection:Disconnect()
                loopFling2Connection = nil
            end
            local myChar = LocalPlayer.Character
            local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHrp then
                myHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
                myHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end
            loopFling2Target = nil
            WindUI:Notify({Title = "循环甩飞", Content = "已关闭", Duration = 1.5, Icon = "check"})
        end
    end})

    fling2Section:Button({Title = "甩飞所有玩家", Callback = function()
        local count = 0
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then
                local char = p.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        teleportFling(p)
                        count = count + 1
                        task.wait(0.2)
                    end
                end
            end
        end
        WindUI:Notify({Title = "甩飞所有玩家", Content = "已甩飞 " .. count .. " 名玩家", Duration = 2, Icon = "check"})
    end})

    local loopAllFling2Running = false
    local loopAllFling2Connection = nil
    fling2Section:Toggle({Title = "循环甩飞所有玩家", Value = false, Callback = function(val)
        if val then
            loopAllFling2Running = true
            WindUI:Notify({Title = "循环甩飞所有玩家", Content = "已开启", Duration = 2, Icon = "check"})

            loopAllFling2Connection = RunService.Heartbeat:Connect(function()
                if not loopAllFling2Running then
                    loopAllFling2Connection:Disconnect()
                    loopAllFling2Connection = nil
                    return
                end
                local players = Players:GetPlayers()
                for _, p in ipairs(players) do
                    if not loopAllFling2Running then break end
                    if p ~= LocalPlayer then
                        local char = p.Character
                        if char then
                            local hum = char:FindFirstChildOfClass("Humanoid")
                            if hum and hum.Health > 0 then
                                teleportFling(p)
                                task.wait(0.2)
                            end
                        end
                    end
                end
                task.wait(0.3)
            end)
        else
            loopAllFling2Running = false
            if loopAllFling2Connection then
                loopAllFling2Connection:Disconnect()
                loopAllFling2Connection = nil
            end
            WindUI:Notify({Title = "循环甩飞所有玩家", Content = "已关闭", Duration = 1.5, Icon = "check"})
        end
    end})

    return tab
