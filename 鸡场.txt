return function(Window, WindUI)
    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Workspace = game:GetService("Workspace")
    local player = Players.LocalPlayer

    local config = {
        eggEnabled = false,
        eggRadius = 40,
        eggAutoDeposit = true,
        eggDepositInterval = 60,
        cashEnabled = false,
        cashDelay = 0.5,
        sellerEnabled = false,
        sellerDelay = 0.5,
        buyEnabled = false,
        buyAmount = 1,
        buyDelay = 0.5,
        mergeEnabled = false,
        mergeDelay = 0.5,
    }

    local eggCollected = 0
    local eggDeposited = 0
    local cashCollected = 0
    local totalCash = 0
    local sellerUpgrades = 0
    local sellerNoMoney = false
    local buyCount = 0
    local buyNoMoney = false
    local mergeCount = 0
    local mergeNoChickens = false
    local isDepositing = false
    local processedEggs = {}
    local eggRunning = false
    local cashRunning = false
    local sellerRunning = false
    local buyRunning = false
    local mergeRunning = false
    local lastDepositTime = 0
    local lastBuyTime = 0
    local lastMergeTime = 0

    local CollectEvent = nil
    local RemoteFunction = nil

    local Paper = ReplicatedStorage:FindFirstChild("Paper")
    if Paper then
        local Remotes = Paper:FindFirstChild("Remotes")
        if Remotes then
            CollectEvent = Remotes:FindFirstChild("__remoteevent")
            RemoteFunction = Remotes:FindFirstChild("__remotefunction")
        end
    end

    if not CollectEvent then
        return
    end

    local function getCharacter()
        return player.Character
    end

    local function getRootPart()
        local char = getCharacter()
        if char then
            return char:FindFirstChild("HumanoidRootPart")
        end
        return nil
    end

    local function getPlayerPosition()
        local root = getRootPart()
        if root then
            return root.Position
        end
        return nil
    end

    local function findNearbyEggs()
        local eggs = {}
        local pos = getPlayerPosition()
        if not pos then return eggs end
        local eggsContainer = Workspace:FindFirstChild("Eggs")
        if not eggsContainer then return eggs end
        for _, obj in ipairs(eggsContainer:GetChildren()) do
            if obj:IsA("Model") then
                local eggId = obj.Name
                local tier = obj:GetAttribute("Tier")
                if tier and not processedEggs[eggId] then
                    local eggPos = nil
                    local primary = obj:FindFirstChild("PrimaryPart")
                    if primary then eggPos = primary.Position end
                    if not eggPos then
                        local root = obj:FindFirstChild("HumanoidRootPart")
                        if root then eggPos = root.Position end
                    end
                    if not eggPos then
                        eggPos = obj:GetPivot().Position
                    end
                    if eggPos then
                        local distance = (eggPos - pos).Magnitude
                        if distance <= config.eggRadius then
                            table.insert(eggs, {
                                Object = obj,
                                Id = eggId,
                                Position = eggPos,
                                Tier = tier,
                                Distance = distance
                            })
                        end
                    end
                end
            end
        end
        table.sort(eggs, function(a, b) return a.Distance < b.Distance end)
        return eggs
    end

    local function teleportToEgg(pos)
        local root = getRootPart()
        if not root then return false end
        local safePos = Vector3.new(pos.X, pos.Y + 3, pos.Z)
        pcall(function()
            root.CFrame = CFrame.new(safePos)
        end)
        task.wait(0.2)
        return true
    end

    local function collectEgg(egg)
        if not egg or not egg.Id then return false end
        if processedEggs[egg.Id] then return false end
        processedEggs[egg.Id] = true
        pcall(function()
            CollectEvent:FireServer("Collect Egg", egg.Id)
            eggCollected = eggCollected + 1
            return true
        end)
        return true
    end

    local function depositEggs()
        if not RemoteFunction or isDepositing then return false end
        isDepositing = true
        local success = false
        pcall(function()
            local result = RemoteFunction:InvokeServer("Deposit Eggs")
            if type(result) == "table" then
                success = result[1] or false
            else
                success = result or false
            end
        end)
        isDepositing = false
        if success then
            eggDeposited = eggDeposited + 1
            lastDepositTime = tick()
        end
        return success
    end

    local function collectCash()
        if not RemoteFunction then return false end
        local success = false
        local amount = 0
        pcall(function()
            local result = RemoteFunction:InvokeServer("Collect Cash")
            if type(result) == "table" then
                success = result[1] or false
                amount = result[2] or 0
            else
                success = result or false
            end
        end)
        if success then
            cashCollected = cashCollected + 1
            totalCash = totalCash + amount
        end
        return success
    end

    local function upgradeSeller()
        if not RemoteFunction then return false end
        local success = false
        local message = ""
        pcall(function()
            local result = RemoteFunction:InvokeServer("Upgrade Process Level")
            if type(result) == "table" then
                success = result[1] or false
                message = result[2] or ""
            else
                success = result or false
            end
        end)
        if success then
            sellerUpgrades = sellerUpgrades + 1
            return true
        else
            if message and message ~= "" then
                if message:find("need") or message:find("afford") then
                    sellerNoMoney = true
                end
            end
            return false
        end
    end

    local function buyChickens()
        if not RemoteFunction then return false end
        local now = tick()
        if now - lastBuyTime < config.buyDelay then
            return false
        end
        local success = false
        local message = ""
        pcall(function()
            local result = RemoteFunction:InvokeServer("Buy Chickens", config.buyAmount)
            if type(result) == "table" then
                success = result[1] or false
                message = result[2] or ""
            else
                success = result or false
            end
        end)
        lastBuyTime = now
        if success then
            buyCount = buyCount + 1
            return true
        else
            if message and message ~= "" then
                if message:find("need") or message:find("afford") or message:find("money") then
                    buyNoMoney = true
                end
            end
            return false
        end
    end

    local function mergeChickens()
        if not RemoteFunction then return false end
        local now = tick()
        if now - lastMergeTime < config.mergeDelay then
            return false
        end
        local success = false
        local message = ""
        pcall(function()
            local result = RemoteFunction:InvokeServer("Merge Chickens")
            if type(result) == "table" then
                success = result[1] or false
                message = result[2] or ""
            else
                success = result or false
            end
        end)
        lastMergeTime = now
        if success then
            mergeCount = mergeCount + 1
            return true
        else
            if message and message ~= "" then
                if message:find("need") or message:find("enough") or message:find("chicken") then
                    mergeNoChickens = true
                end
            end
            return false
        end
    end

    local function eggLoop()
        local noEggsCount = 0
        lastDepositTime = tick()
        while eggRunning do
            task.wait(0.1)
            if not config.eggEnabled then
                task.wait(0.5)
                continue
            end
            if config.eggAutoDeposit then
                local timeSinceDeposit = tick() - lastDepositTime
                if timeSinceDeposit >= config.eggDepositInterval then
                    if config.eggEnabled then
                        depositEggs()
                    end
                end
            end
            local eggs = findNearbyEggs()
            if #eggs > 0 then
                noEggsCount = 0
                for _, egg in ipairs(eggs) do
                    if not eggRunning or not config.eggEnabled then break end
                    if not egg.Object or not egg.Object.Parent then continue end
                    teleportToEgg(egg.Position)
                    task.wait(0.2)
                    collectEgg(egg)
                    task.wait(0.2)
                end
                task.wait(1)
            else
                noEggsCount = noEggsCount + 1
                if noEggsCount >= 3 and config.eggAutoDeposit then
                    depositEggs()
                    noEggsCount = 0
                    task.wait(2)
                else
                    task.wait(2)
                end
            end
        end
    end

    local function cashLoop()
        while cashRunning do
            task.wait(0.1)
            if not config.cashEnabled then
                task.wait(0.5)
                continue
            end
            collectCash()
            task.wait(config.cashDelay)
        end
    end

    local function sellerLoop()
        sellerNoMoney = false
        while sellerRunning do
            task.wait(0.1)
            if not config.sellerEnabled or sellerNoMoney then
                if sellerNoMoney then
                    task.wait(5)
                end
                task.wait(0.5)
                continue
            end
            upgradeSeller()
            task.wait(config.sellerDelay)
        end
    end

    local function buyLoop()
        buyNoMoney = false
        lastBuyTime = 0
        while buyRunning do
            task.wait(0.1)
            if not config.buyEnabled or buyNoMoney then
                if buyNoMoney then
                    task.wait(5)
                end
                task.wait(0.5)
                continue
            end
            buyChickens()
            task.wait(config.buyDelay)
        end
    end

    local function mergeLoop()
        mergeNoChickens = false
        lastMergeTime = 0
        while mergeRunning do
            task.wait(0.1)
            if not config.mergeEnabled or mergeNoChickens then
                if mergeNoChickens then
                    task.wait(5)
                end
                task.wait(0.5)
                continue
            end
            mergeChickens()
            task.wait(config.mergeDelay)
        end
    end

    local function toggleEgg(state)
        config.eggEnabled = state
        if state then
            lastDepositTime = tick()
            if not eggRunning then
                eggRunning = true
                task.spawn(eggLoop)
            end
        end
    end

    local function toggleCash(state)
        config.cashEnabled = state
        if state then
            if not cashRunning then
                cashRunning = true
                task.spawn(cashLoop)
            end
        end
    end

    local function toggleSeller(state)
        config.sellerEnabled = state
        if state then
            if not sellerRunning then
                sellerRunning = true
                sellerNoMoney = false
                task.spawn(sellerLoop)
            end
        end
    end

    local function toggleBuy(state)
        config.buyEnabled = state
        if state then
            buyNoMoney = false
            if not buyRunning then
                buyRunning = true
                task.spawn(buyLoop)
            end
        end
    end

    local function toggleMerge(state)
        config.mergeEnabled = state
        if state then
            mergeNoChickens = false
            if not mergeRunning then
                mergeRunning = true
                task.spawn(mergeLoop)
            end
        end
    end

    local function setBuyAmount(amount)
        local validAmounts = {1, 5, 25, 100}
        local valid = false
        for _, v in ipairs(validAmounts) do
            if v == amount then
                valid = true
                break
            end
        end
        if not valid then
            return
        end
        config.buyAmount = amount
    end

    local function manualDeposit()
        depositEggs()
    end

    local tab = Window:Tab({Title = "鸡场", Icon = ""})

    local eggSection = tab:Section({Title = "蛋"})

    eggSection:Toggle({
        Title = "自动捡蛋",
        Value = false,
        Callback = toggleEgg
    })

    eggSection:Slider({
        Title = "捡蛋范围",
        Value = {Min = 10, Max = 80, Default = 40},
        Callback = function(value)
            config.eggRadius = math.floor(value)
        end
    })

    eggSection:Toggle({
        Title = "自动存蛋",
        Value = true,
        Callback = function(state)
            config.eggAutoDeposit = state
            if state then lastDepositTime = tick() end
        end
    })

    eggSection:Slider({
        Title = "存蛋间隔",
        Value = {Min = 10, Max = 300, Default = 60},
        Callback = function(value)
            config.eggDepositInterval = math.floor(value)
        end
    })

    eggSection:Button({
        Title = "手动存蛋",
        Callback = manualDeposit
    })

    tab:Divider()

    local cashSection = tab:Section({Title = "金币"})

    cashSection:Toggle({
        Title = "自动收钱",
        Value = false,
        Callback = toggleCash
    })

    cashSection:Slider({
        Title = "收钱间隔",
        Value = {Min = 0.1, Max = 2, Default = 0.5},
        Callback = function(value)
            config.cashDelay = value
        end
    })

    tab:Divider()

    local sellerSection = tab:Section({Title = "出售"})

    sellerSection:Toggle({
        Title = "自动升级出售",
        Value = false,
        Callback = toggleSeller
    })

    sellerSection:Slider({
        Title = "升级间隔",
        Value = {Min = 0.2, Max = 2, Default = 0.5},
        Callback = function(value)
            config.sellerDelay = value
        end
    })

    tab:Divider()

    local chickenSection = tab:Section({Title = "鸡"})

    chickenSection:Toggle({
        Title = "自动买鸡",
        Value = false,
        Callback = toggleBuy
    })

    chickenSection:Slider({
        Title = "买鸡数量",
        Value = {Min = 1, Max = 100, Default = 1},
        Callback = function(value)
            config.buyAmount = math.floor(value)
        end
    })

    chickenSection:Slider({
        Title = "买鸡间隔",
        Value = {Min = 0.2, Max = 2, Default = 0.5},
        Callback = function(value)
            config.buyDelay = value
        end
    })

    chickenSection:Toggle({
        Title = "自动合鸡",
        Value = false,
        Callback = toggleMerge
    })

    chickenSection:Slider({
        Title = "合鸡间隔",
        Value = {Min = 0.2, Max = 2, Default = 0.5},
        Callback = function(value)
            config.mergeDelay = value
        end
    })

    return tab
end
