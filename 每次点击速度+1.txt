return function(Window, WindUI)
    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Workspace = game:GetService("Workspace")
    local LP = Players.LocalPlayer

    local tab = Window:Tab({Title = "每次点击速度+1", Icon = "zap"})

    local energyThread = nil
    local coinThread = nil
    local infEnergyActive = false
    local infCoinActive = false

    local function GetStats()
        return LP:FindFirstChild("leaderstats")
    end

    local function StartEnergy()
        if energyThread then task.cancel(energyThread) end
        infEnergyActive = true
        energyThread = task.spawn(function()
            while infEnergyActive do
                task.wait(0.04)
                local stats = GetStats()
                if stats then
                    local energyObj = stats:FindFirstChild("Energy") or stats:FindFirstChild("Speed") or stats:FindFirstChild("Power")
                    if energyObj and energyObj:IsA("ValueBase") then
                        pcall(function() energyObj.Value = energyObj.Value + 9999 end)
                    end
                end
                for k, v in pairs(LP:GetAttributes()) do
                    local lname = string.lower(tostring(k))
                    if string.find(lname, "energy") or string.find(lname, "speed") or string.find(lname, "power") then
                        if type(v) == "number" then
                            pcall(function() LP:SetAttribute(k, v + 9999) end)
                        end
                    end
                end
                local pgui = LP:FindFirstChild("PlayerGui")
                for _, container in ipairs({ ReplicatedStorage, pgui, Workspace }) do
                    if container then
                        for _, item in ipairs(container:GetDescendants()) do
                            if item:IsA("RemoteEvent") or item:IsA("RemoteFunction") then
                                local lname = string.lower(item.Name)
                                if string.find(lname, "energy") or string.find(lname, "click") or string.find(lname, "speed") or string.find(lname, "power") or string.find(lname, "tap") or string.find(lname, "add") then
                                    pcall(function()
                                        if item:IsA("RemoteEvent") then
                                            item:FireServer(9999)
                                            item:FireServer()
                                            item:FireServer("Energy", 9999)
                                        else
                                            item:InvokeServer(9999)
                                            item:InvokeServer()
                                        end
                                    end)
                                end
                            end
                        end
                    end
                end
                local char = LP.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if root and firetouchinterest then
                    for _, obj in ipairs(Workspace:GetDescendants()) do
                        if obj:IsA("TouchTransmitter") then
                            local part = obj.Parent
                            if part and part:IsA("BasePart") then
                                local pname = string.lower(part.Name)
                                if string.find(pname, "energy") or string.find(pname, "speed") or string.find(pname, "orb") then
                                    pcall(function()
                                        firetouchinterest(part, root, 0)
                                        task.wait()
                                        firetouchinterest(part, root, 1)
                                    end)
                                end
                            end
                        end
                    end
                end
            end
        end)
    end

    local function StopEnergy()
        infEnergyActive = false
        if energyThread then task.cancel(energyThread); energyThread = nil end
    end

    local function StartCoin()
        if coinThread then task.cancel(coinThread) end
        infCoinActive = true
        coinThread = task.spawn(function()
            while infCoinActive do
                task.wait(0.04)
                local stats = GetStats()
                if stats then
                    local coinObj = stats:FindFirstChild("Coin") or stats:FindFirstChild("Coins") or stats:FindFirstChild("Money") or stats:FindFirstChild("Gold")
                    if coinObj and coinObj:IsA("ValueBase") then
                        pcall(function() coinObj.Value = coinObj.Value + 9999 end)
                    end
                end
                for k, v in pairs(LP:GetAttributes()) do
                    local lname = string.lower(tostring(k))
                    if string.find(lname, "coin") or string.find(lname, "money") or string.find(lname, "gold") then
                        if type(v) == "number" then
                            pcall(function() LP:SetAttribute(k, v + 9999) end)
                        end
                    end
                end
                local pgui = LP:FindFirstChild("PlayerGui")
                for _, container in ipairs({ ReplicatedStorage, pgui, Workspace }) do
                    if container then
                        for _, item in ipairs(container:GetDescendants()) do
                            if item:IsA("RemoteEvent") or item:IsA("RemoteFunction") then
                                local lname = string.lower(item.Name)
                                if string.find(lname, "coin") or string.find(lname, "money") or string.find(lname, "gold") or string.find(lname, "reward") or string.find(lname, "claim") or string.find(lname, "cash") then
                                    pcall(function()
                                        if item:IsA("RemoteEvent") then
                                            item:FireServer(9999)
                                            item:FireServer()
                                            item:FireServer("Coin", 9999)
                                        else
                                            item:InvokeServer(9999)
                                            item:InvokeServer()
                                        end
                                    end)
                                end
                            end
                        end
                    end
                end
                local char = LP.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if root and firetouchinterest then
                    for _, obj in ipairs(Workspace:GetDescendants()) do
                        if obj:IsA("TouchTransmitter") then
                            local part = obj.Parent
                            if part and part:IsA("BasePart") then
                                local pname = string.lower(part.Name)
                                if string.find(pname, "coin") or string.find(pname, "gold") or string.find(pname, "money") or string.find(pname, "cash") then
                                    pcall(function()
                                        firetouchinterest(part, root, 0)
                                        task.wait()
                                        firetouchinterest(part, root, 1)
                                    end)
                                end
                            end
                        end
                    end
                end
            end
        end)
    end

    local function StopCoin()
        infCoinActive = false
        if coinThread then task.cancel(coinThread); coinThread = nil end
    end

    tab:Toggle({
        Title = "无限能量",
        Value = false,
        Callback = function(val)
            if val then
                StartEnergy()
            else
                StopEnergy()
            end
        end
    })

    tab:Toggle({
        Title = "无限金币",
        Value = false,
        Callback = function(val)
            if val then
                StartCoin()
            else
                StopCoin()
            end
        end
    })

    return tab
end
