return function(Window, WindUI)
    local player = game.Players.LocalPlayer

    local spinsLock = false
    local spinsLockConn = nil

    local tab = Window:Tab({Title = "每点击+1年龄", Icon = ""})

    tab:Toggle({
        Title = "锁定转盘次数5",
        Value = false,
        Callback = function(state)
            spinsLock = state

            if state then
                local spins = player:FindFirstChild("Spins")
                if spins then
                    spins.Value = 5
                end

                spinsLockConn = spins:GetPropertyChangedSignal("Value"):Connect(function()
                    if spinsLock and spins.Value ~= 5 then
                        spins.Value = 5
                    end
                end)
            else
                if spinsLockConn then
                    spinsLockConn:Disconnect()
                    spinsLockConn = nil
                end
            end
        end
    })

    return tab
end
