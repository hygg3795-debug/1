return function(Window, WindUI)
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local Workspace = game:GetService("Workspace")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local MarketplaceService = game:GetService("MarketplaceService")
    local LocalPlayer = Players.LocalPlayer

    local tab = Window:Tab({Title = "+1键盘速度逃脱", Icon = " "})

    local WinPos = Vector3.new(-6809.3223, 531.2539, 1468.8073)

    local autoRebirthLoop = nil
    local autoRebirthEnabled = false

    local invincibleEnabled = false
    local healthConnection = nil
    local hazardConnection = nil

    local autoWinEnabled = false
    local autoWinLoop = nil

    local hazardKeywords = {
        "kill", "lava", "dead", "wave", "wall", "crush", "water", "flood"
    }

    local function isHazard(name)
        local nameLower = name:lower()
        for _, keyword in ipairs(hazardKeywords) do
            if nameLower:match(keyword) then return true end
        end
        return false
    end

    local function bypassLocalTraps(object)
        if not invincibleEnabled then return end
        if object:IsA("BasePart") and isHazard(object.Name) then
            local touchInterest = object:FindFirstChildOfClass("TouchTransmitter")
            if touchInterest then pcall(function() touchInterest:Destroy() end) end
        elseif object:IsA("TouchTransmitter") and object.Parent and isHazard(object.Parent.Name) then
            task.wait()
            pcall(function() object:Destroy() end)
        end
    end

    local function startHealthLock()
        if healthConnection then healthConnection:Disconnect() healthConnection = nil end
        if not invincibleEnabled then return end
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            healthConnection = hum.HealthChanged:Connect(function(health)
                if invincibleEnabled and health < hum.MaxHealth and health > 0 then
                    hum.Health = hum.MaxHealth
                end
            end)
        end
    end

    local function stopHealthLock()
        if healthConnection then
            healthConnection:Disconnect()
            healthConnection = nil
        end
    end

    LocalPlayer.CharacterAdded:Connect(function()
        task.wait(0.3)
        if invincibleEnabled then
            startHealthLock()
            for _, item in ipairs(Workspace:GetDescendants()) do bypassLocalTraps(item) end
        end
    end)

    local function isCharacterValid()
        local char = LocalPlayer.Character
        if not char then return false end
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return false end
        if hum.Health <= 0 then return false end
        return true
    end

    local function startAutoWinLoop()
        if autoWinLoop then task.cancel(autoWinLoop) end
        autoWinLoop = task.spawn(function()
            while autoWinEnabled do
                if isCharacterValid() then
                    local char = LocalPlayer.Character
                    local root = char:FindFirstChild("HumanoidRootPart")
                    local hum = char:FindFirstChildOfClass("Humanoid")

                    if root and hum then
                        local dist = (root.Position - WinPos).Magnitude

                        if dist > 5 then
                            local originalWalkSpeed = hum.WalkSpeed
                            hum.WalkSpeed = 0

                            local riseHeight = 900
                            local riseStart = root.CFrame
                            local riseEnd = riseStart * CFrame.new(0, riseHeight, 0)

                            for i = 0, 1, 0.05 do
                                if not autoWinEnabled or not root.Parent or hum.Health <= 0 then break end
                                root.CFrame = riseStart:Lerp(riseEnd, i)
                                task.wait(0.02)
                            end

                            if not autoWinEnabled or not root.Parent or hum.Health <= 0 then
                                hum.WalkSpeed = originalWalkSpeed
                                break
                            end
                            task.wait(0.2)

                            local startCF = root.CFrame
                            local targetPos = Vector3.new(WinPos.X, root.Position.Y, WinPos.Z)
                            local targetCF = CFrame.new(targetPos, targetPos + Vector3.new(0, 0, 1))

                            local distance = (root.Position - targetPos).Magnitude
                            local steps = math.floor(distance / (300 / 30))
                            if steps < 1 then steps = 1 end

                            for i = 1, steps do
                                if not autoWinEnabled or not root.Parent or hum.Health <= 0 then break end
                                root.CFrame = startCF:Lerp(targetCF, i / steps)
                                task.wait(0.03)
                            end

                            if not autoWinEnabled or not root.Parent or hum.Health <= 0 then
                                hum.WalkSpeed = originalWalkSpeed
                                break
                            end
                            task.wait(0.2)

                            local currentCF = root.CFrame
                            local finalCF = CFrame.new(WinPos)

                            for i = 0, 1, 0.01 do
                                if not autoWinEnabled or not root.Parent or hum.Health <= 0 then break end
                                root.CFrame = currentCF:Lerp(finalCF, i)
                                task.wait(0.01)
                            end

                            if root.Parent and hum.Health > 0 then
                                root.CFrame = finalCF
                            end

                            hum.WalkSpeed = originalWalkSpeed

                            if root.Parent and hum.Health > 0 then
                                task.wait(1)
                                hum.Health = 0
                                task.wait(0.5)
                            end
                        end
                    end
                end
                task.wait(0.1)
            end
        end)
    end

    tab:Toggle({
        Title = "自动重生",
        Value = false,
        Callback = function(val)
            autoRebirthEnabled = val
            if autoRebirthLoop then
                task.cancel(autoRebirthLoop)
                autoRebirthLoop = nil
            end
            if val then
                autoRebirthLoop = task.spawn(function()
                    while autoRebirthEnabled do
                        pcall(function()
                            local remotes = ReplicatedStorage:FindFirstChild("Remotes") or ReplicatedStorage
                            for _, r in ipairs(remotes:GetDescendants()) do
                                if r:IsA("RemoteEvent") and (r.Name:find("Rebirth") or r.Name:find("Prestige")) then
                                    r:FireServer()
                                end
                            end
                        end)
                        task.wait(1)
                    end
                end)
            end
        end
    })

    tab:Toggle({
        Title = "无敌",
        Value = false,
        Callback = function(val)
            invincibleEnabled = val

            if hazardConnection then
                hazardConnection:Disconnect()
                hazardConnection = nil
            end

            if invincibleEnabled then
                hazardConnection = Workspace.DescendantAdded:Connect(bypassLocalTraps)
                for _, item in ipairs(Workspace:GetDescendants()) do bypassLocalTraps(item) end
                startHealthLock()
            else
                stopHealthLock()
            end
        end
    })

    tab:Toggle({
        Title = "自动胜利",
        Value = false,
        Callback = function(val)
            autoWinEnabled = val
            if val then
                startAutoWinLoop()
            else
                if autoWinLoop then
                    task.cancel(autoWinLoop)
                    autoWinLoop = nil
                end
            end
        end
    })

    tab:Button({
        Title = "瞬移到胜利点",
        Icon = "target",
        Justify = "Center",
        Color = Color3.fromHex("#6C5CE7"),
        Callback = function()
            local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.CFrame = CFrame.new(WinPos) * CFrame.new(0, 5, 0)
                WindUI:Notify({
                    Title = "传送",
                    Content = "已瞬移到胜利点!",
                    Icon = "target",
                    Duration = 2
                })
            else
                WindUI:Notify({
                    Title = "传送失败",
                    Content = "角色不存在!",
                    Icon = "alert-circle",
                    Duration = 2
                })
            end
        end
    })

    tab:Button({
        Title = "解锁本地音效",
        Icon = "music",
        Justify = "Center",
        Color = Color3.fromHex("#34D399"),
        Callback = function()
            local ids = { 1928718370, 1724877043 }
            for _, id in ipairs(ids) do
                pcall(function()
                    MarketplaceService:SignalPromptGamePassPurchaseFinished(LocalPlayer, id, true)
                end)
                pcall(function()
                    MarketplaceService:SignalPromptPurchaseFinished(LocalPlayer, id, true)
                end)
                pcall(function()
                    MarketplaceService:SignalPromptProductPurchaseFinished(LocalPlayer, id, true)
                end)
            end
            if hookfunction then
                local oldOwns
                oldOwns = hookfunction(MarketplaceService.UserOwnsGamePassAsync, function(self, p, passId)
                    if typeof(passId) == "number" and table.find(ids, passId) and p == LocalPlayer then
                        return true
                    end
                    return oldOwns(self, p, passId)
                end)
            end
            WindUI:Notify({
                Title = "音效解锁",
                Content = "音效已解锁!",
                Icon = "music",
                Duration = 3
            })
        end
    })

    return tab
end
