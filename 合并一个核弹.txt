return function(Window, WindUI)
    local cloneref = (cloneref or clonereference or function(i) return i end)

    local Players = cloneref(game:GetService("Players"))
    local TeleportService = cloneref(game:GetService("TeleportService"))
    local RunService = cloneref(game:GetService("RunService"))
    local ReplicatedStorage = cloneref(game:GetService("ReplicatedStorage"))
    local Workspace = cloneref(game:GetService("Workspace"))

    local LocalPlayer = Players.LocalPlayer

    local NukeRemotes = ReplicatedStorage:WaitForChild("NukeRemotes", 10)
    local R_PickUp = NukeRemotes and NukeRemotes:WaitForChild("拾取", 5)
    local R_Drop = NukeRemotes and NukeRemotes:WaitForChild("丢出", 5)
    local R_MergeRequest = NukeRemotes and NukeRemotes:WaitForChild("合并请求", 5)
    local R_HoldStarted = NukeRemotes and NukeRemotes:WaitForChild("持有开始", 5)
    local R_HoldEnded = NukeRemotes and NukeRemotes:WaitForChild("持有结束", 5)
    local R_RequestLock = NukeRemotes and NukeRemotes:WaitForChild("请求锁定基地", 5)
    local R_LockStateUpdate = NukeRemotes and NukeRemotes:WaitForChild("锁定状态更新", 5)
    local R_PurchaseUpgrade = NukeRemotes and NukeRemotes:WaitForChild("购买升级", 5)
    local R_RequestRebirth = NukeRemotes and NukeRemotes:WaitForChild("请求重生", 5)
    local R_RedeemCode = NukeRemotes and NukeRemotes:WaitForChild("兑换码", 5)

    local DROP_DEBOUNCE = 0.75
    local DROP_ESCAPE_DIST = 15
    local DROP_ZONE_RADIUS = 10
    local DROP_ZONE_EXPIRE = 4
    local CLOSE_NUKE_THRESHOLD = 4
    local SEPARATE_DIST = 18

    local serverHeldTier = nil
    local holdStartedConn, holdEndedConn
    local recentDrops = {}

    holdStartedConn = R_HoldStarted and R_HoldStarted.OnClientEvent:Connect(function(等级)
        serverHeldTier = 等级
    end)

    holdEndedConn = R_HoldEnded and R_HoldEnded.OnClientEvent:Connect(function()
        serverHeldTier = nil
    end)

    local lockPhase = "free"
    local lockSecondsLeft = 0

    if R_LockStateUpdate then
        R_LockStateUpdate.OnClientEvent:Connect(function(状态, 秒数)
            lockPhase = 状态 or "free"
            lockSecondsLeft = 秒数 or 0
        end)
    end

    local State = {
        自动合并启用 = false,
        自动合并线程 = nil,
        合并延迟 = 0.2,
        统计 = { 合并次数 = 0, 丢出次数 = 0, 错误次数 = 0 },
        最高等级 = 0,
        自动锁定启用 = false,
        自动锁定线程 = nil,
        自动离开启用 = false,
        自动离开线程 = nil,
        重连延迟 = 0,
        检测半径 = 50,
        智能锁定启用 = false,
        智能锁定线程 = nil,
        智能锁定半径 = 90,
        自动重生启用 = false,
        自动重生线程 = nil,
        自动重连启用 = false,
        自动重连线程 = nil,
        自动重连分钟 = 20,
        防挂机启用 = false,
        防挂机线程 = nil,
        自动升级 = {
            等级 = false,
            最大 = false,
            锁定基地 = false,
        },
        自动升级线程 = {},
    }

    local function 获取角色()
        return LocalPlayer.Character
    end

    local function 获取人体根部()
        local c = 获取角色()
        return c and c:FindFirstChild("HumanoidRootPart")
    end

    local function 获取人性()
        local c = 获取角色()
        return c and c:FindFirstChildWhichIsA("Humanoid")
    end

    local cachedNukesFolder = nil

    local function 获取我的核弹文件夹()
        if cachedNukesFolder and cachedNukesFolder.Parent then
            return cachedNukesFolder
        end
        cachedNukesFolder = nil
        local bases = Workspace:FindFirstChild("Bases")
        if not bases then
            return nil
        end
        for _, base in ipairs(bases:GetChildren()) do
            local nukes = base:FindFirstChild("Nukes")
            if nukes then
                for _, nuke in ipairs(nukes:GetChildren()) do
                    if nuke:IsA("BasePart") and nuke:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                        cachedNukesFolder = nukes
                        return nukes
                    end
                end
            end
        end
        return nil
    end

    local function XZ距离(a, b)
        local dx = a.X - b.X
        local dz = a.Z - b.Z
        return math.sqrt(dx * dx + dz * dz)
    end

    local function 传送至(位置)
        local hrp = 获取人体根部()
        if not hrp then
            return
        end
        hrp.CFrame = CFrame.new(位置 + Vector3.new(0, 4, 2))
    end

    local function 获取岛屿中心()
        local folder = 获取我的核弹文件夹()
        if not folder then
            return Vector3.new(0, 0, 0)
        end
        local sum, count = Vector3.new(0, 0, 0), 0
        for _, part in ipairs(folder:GetChildren()) do
            if part:IsA("BasePart") and part:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                sum = sum + part.Position
                count = count + 1
            end
        end
        return count > 0 and (sum / count) or Vector3.new(0, 0, 0)
    end

    local function 清除过期丢出记录()
        local now = tick()
        for i = #recentDrops, 1, -1 do
            if now - recentDrops[i].time > DROP_ZONE_EXPIRE then
                table.remove(recentDrops, i)
            end
        end
    end

    local function 记录丢出(等级)
        local hrp = 获取人体根部()
        if not hrp then
            return
        end
        清除过期丢出记录()
        table.insert(recentDrops, { pos = hrp.Position, tier = 等级, time = tick() })
    end

    local function 是否靠近最近丢出位置(世界位置, 忽略等级)
        清除过期丢出记录()
        for _, e in ipairs(recentDrops) do
            if e.tier ~= 忽略等级 then
                if XZ距离(世界位置, e.pos) < DROP_ZONE_RADIUS then
                    return true
                end
            end
        end
        return false
    end

    local function 逃离丢出区域(丢出位置, 目标位置)
        local hrp = 获取人体根部()
        if not hrp then
            return
        end
        if 目标位置 and XZ距离(目标位置, 丢出位置) >= DROP_ZONE_RADIUS then
            return
        end
        local myPos = hrp.Position
        local dx = myPos.X - 丢出位置.X
        local dz = myPos.Z - 丢出位置.Z
        local len = math.sqrt(dx * dx + dz * dz)
        if len < 0.1 then
            dx, dz, len = 1, 0, 1
        end
        dx, dz = dx / len, dz / len
        传送至(Vector3.new(
            丢出位置.X + dx * DROP_ESCAPE_DIST,
            丢出位置.Y,
            丢出位置.Z + dz * DROP_ESCAPE_DIST
        ))
        task.wait(DROP_DEBOUNCE + 0.15)
    end

    local function 按等级获取我的核弹()
        local folder = 获取我的核弹文件夹()
        if not folder then
            return {}
        end
        清除过期丢出记录()
        local byTier = {}
        for _, part in ipairs(folder:GetChildren()) do
            if part:IsA("BasePart") and part:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                local 等级 = part:GetAttribute("Tier")
                local 状态 = part:GetAttribute("State")
                if 等级 and (状态 == "floor" or 状态 == "based" or 状态 == nil) then
                    if not byTier[等级] then
                        byTier[等级] = {}
                    end
                    if 是否靠近最近丢出位置(part.Position, 等级) then
                        table.insert(byTier[等级], part)
                    else
                        table.insert(byTier[等级], 1, part)
                    end
                end
            end
        end
        return byTier
    end

    local function 分离核弹(要移动的核弹, 参考位置)
        if serverHeldTier ~= nil then
            return
        end
        if not 要移动的核弹 or not 要移动的核弹.Parent then
            return
        end
        传送至(要移动的核弹.Position)
        task.wait(0.15)
        pcall(function()
            R_PickUp:FireServer(要移动的核弹)
        end)
        local t = tick()
        repeat
            task.wait(0.05)
        until serverHeldTier ~= nil or (tick() - t) > 1.5
        if serverHeldTier == nil then
            return
        end
        local hrp = 获取人体根部()
        if not hrp then
            pcall(function()
                R_Drop:FireServer()
            end)
            serverHeldTier = nil
            return
        end
        local awayX = hrp.Position.X - 参考位置.X
        local awayZ = hrp.Position.Z - 参考位置.Z
        local awayLen = math.sqrt(awayX * awayX + awayZ * awayZ)
        if awayLen < 0.1 then
            awayX, awayZ, awayLen = 1, 0, 1
        end
        awayX, awayZ = awayX / awayLen, awayZ / awayLen
        local dropPos = Vector3.new(
            参考位置.X + awayX * SEPARATE_DIST,
            参考位置.Y,
            参考位置.Z + awayZ * SEPARATE_DIST
        )
        传送至(dropPos)
        task.wait(0.1)
        local droppedTier = serverHeldTier
        local hrp2 = 获取人体根部()
        local actualDrop = hrp2 and hrp2.Position or dropPos
        pcall(function()
            R_Drop:FireServer()
        end)
        local t2 = tick()
        repeat
            task.wait(0.05)
        until serverHeldTier == nil or (tick() - t2) > (DROP_DEBOUNCE + 0.5)
        serverHeldTier = nil
        State.统计.丢出次数 = State.统计.丢出次数 + 1
        if droppedTier then
            table.insert(recentDrops, { pos = actualDrop, tier = droppedTier, time = tick() })
        end
        task.wait(DROP_DEBOUNCE + 0.2)
    end

    local function 丢出核弹(目标位置)
        local droppedTier = serverHeldTier
        local hrp = 获取人体根部()
        local droppedPos = hrp and hrp.Position or Vector3.new(0, 0, 0)
        pcall(function()
            R_Drop:FireServer()
        end)
        State.统计.丢出次数 = State.统计.丢出次数 + 1
        local t = tick()
        repeat
            task.wait(0.05)
        until serverHeldTier == nil or (tick() - t) > (DROP_DEBOUNCE + 0.5)
        serverHeldTier = nil
        if droppedTier then
            记录丢出(droppedTier)
        end
        逃离丢出区域(droppedPos, 目标位置)
    end

    local function 拾取核弹(nuke)
        if not nuke or not nuke.Parent then
            return nil
        end
        local expectedTier = nuke:GetAttribute("Tier")
        if not expectedTier then
            return nil
        end
        if serverHeldTier ~= nil then
            return nil
        end
        local folder = 获取我的核弹文件夹()
        if folder then
            for _, other in ipairs(folder:GetChildren()) do
                if other ~= nuke and other:IsA("BasePart") and other:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                    local otherTier = other:GetAttribute("Tier")
                    local otherState = other:GetAttribute("State")
                    if otherTier and otherTier ~= expectedTier and (otherState == "floor" or otherState == "based" or otherState == nil) and XZ距离(nuke.Position, other.Position) <= CLOSE_NUKE_THRESHOLD then
                        分离核弹(other, nuke.Position)
                        task.wait(0.1)
                        break
                    end
                end
            end
        end
        if serverHeldTier ~= nil then
            丢出核弹(nuke.Position)
        end
        传送至(nuke.Position)
        task.wait(0.15)
        if not nuke.Parent then
            return nil
        end
        pcall(function()
            R_PickUp:FireServer(nuke)
        end)
        local t = tick()
        repeat
            task.wait(0.05)
        until serverHeldTier ~= nil or (tick() - t) > 1.5
        if serverHeldTier == nil then
            State.统计.错误次数 = State.统计.错误次数 + 1
            return nil
        end
        return serverHeldTier
    end

    local function 合并到目标(目标, 持有等级)
        if not 目标 or not 目标.Parent then
            return false
        end
        local targetTier = 目标:GetAttribute("Tier")
        if targetTier ~= 持有等级 then
            return false
        end
        传送至(目标.Position)
        task.wait(0.1)
        if not 目标.Parent then
            return false
        end
        local preMergeTier = serverHeldTier
        local ok = pcall(function()
            R_MergeRequest:FireServer(目标)
        end)
        if not ok then
            State.统计.错误次数 = State.统计.错误次数 + 1
            return false
        end
        local t = tick()
        repeat
            task.wait(0.05)
        until serverHeldTier ~= preMergeTier or (tick() - t) > 1.2
        if serverHeldTier == preMergeTier then
            State.统计.错误次数 = State.统计.错误次数 + 1
            return false
        end
        State.统计.合并次数 = State.统计.合并次数 + 1
        if serverHeldTier and serverHeldTier > State.最高等级 then
            State.最高等级 = serverHeldTier
        end
        return true, serverHeldTier
    end

    local function 执行合并循环()
        if serverHeldTier ~= nil then
            local heldTier = serverHeldTier
            local byTier = 按等级获取我的核弹()
            local targets = byTier[heldTier]
            if targets and #targets >= 1 then
                local merged = 合并到目标(targets[1], heldTier)
                if merged then
                    task.wait(0.3)
                else
                    丢出核弹(获取岛屿中心())
                    task.wait(0.3)
                    return
                end
            else
                丢出核弹(获取岛屿中心())
                task.wait(0.3)
                return
            end
        end
        if serverHeldTier ~= nil then
            local heldTier = serverHeldTier
            local byTier = 按等级获取我的核弹()
            local targets = byTier[heldTier]
            if targets and #targets >= 1 then
                local merged = 合并到目标(targets[1], heldTier)
                if merged then
                    task.wait(0.3)
                    return
                else
                    丢出核弹(获取岛屿中心())
                    task.wait(0.3)
                    return
                end
            else
                丢出核弹(获取岛屿中心())
                task.wait(DROP_DEBOUNCE)
                return
            end
        end
        local byTier = 按等级获取我的核弹()
        local mergeableTiers = {}
        for 等级, nukes in pairs(byTier) do
            if #nukes >= 2 then
                table.insert(mergeableTiers, 等级)
            end
        end
        table.sort(mergeableTiers)
        if #mergeableTiers == 0 then
            task.wait(1.5)
            return
        end
        local targetTier = mergeableTiers[1]
        local nukes = byTier[targetTier]
        local nukeA = nukes[1]
        local nukeB = nukes[2]
        if not nukeA or not nukeA.Parent or not nukeB or not nukeB.Parent then
            return
        end
        if nukeA:GetAttribute("Tier") ~= targetTier then
            return
        end
        if nukeB:GetAttribute("Tier") ~= targetTier then
            return
        end
        if serverHeldTier ~= nil then
            local heldNow = serverHeldTier
            local heldValue = 2 ^ (heldNow - 1)
            local pairValue = 2 ^ (targetTier - 1)
            if heldNow == targetTier then
                local merged = 合并到目标(nukeA, heldNow)
                if merged then
                    task.wait(0.3)
                else
                    丢出核弹(获取岛屿中心())
                    task.wait(0.3)
                end
                return
            end
            if heldValue > pairValue then
                丢出核弹(获取岛屿中心())
            else
                丢出核弹(获取岛屿中心())
            end
            task.wait(DROP_DEBOUNCE + 0.1)
            return
        end
        local heldTier = 拾取核弹(nukeA)
        if not heldTier then
            task.wait(0.5)
            return
        end
        if heldTier ~= targetTier then
            task.wait(0.2)
            return
        end
        task.wait(0.1)
        if not nukeB or not nukeB.Parent or nukeB:GetAttribute("Tier") ~= targetTier then
            丢出核弹(获取岛屿中心())
            task.wait(0.3)
            return
        end
        local merged = 合并到目标(nukeB, heldTier)
        if not merged then
            丢出核弹(获取岛屿中心())
            task.wait(0.3)
        else
            task.wait(0.2)
        end
    end

    local function 自动合并循环()
        while State.自动合并启用 do
            local ok, err = pcall(执行合并循环)
            if not ok then
                State.统计.错误次数 = State.统计.错误次数 + 1
                local hrp2 = 获取人体根部()
                local crashPos = hrp2 and hrp2.Position or Vector3.new(0, 0, 0)
                if serverHeldTier then
                    记录丢出(serverHeldTier)
                end
                pcall(function()
                    R_Drop:FireServer()
                end)
                serverHeldTier = nil
                pcall(function()
                    逃离丢出区域(crashPos, 获取岛屿中心())
                end)
            end
            task.wait(State.合并延迟)
        end
        pcall(function()
            R_Drop:FireServer()
        end)
        serverHeldTier = nil
        table.clear(recentDrops)
    end

    local function 启动自动合并()
        if State.自动合并线程 then
            task.cancel(State.自动合并线程)
        end
        pcall(function()
            R_Drop:FireServer()
        end)
        task.wait(DROP_DEBOUNCE + 0.2)
        serverHeldTier = nil
        table.clear(recentDrops)
        State.自动合并启用 = true
        State.自动合并线程 = task.spawn(自动合并循环)
    end

    local function 停止自动合并()
        State.自动合并启用 = false
        if State.自动合并线程 then
            task.cancel(State.自动合并线程)
            State.自动合并线程 = nil
        end
        pcall(function()
            R_Drop:FireServer()
        end)
        serverHeldTier = nil
    end

    local function 自动锁定循环()
        local phaseEnteredAt = tick()
        local trackedPhase = lockPhase
        while State.自动锁定启用 do
            if lockPhase ~= trackedPhase then
                trackedPhase = lockPhase
                phaseEnteredAt = tick()
            end
            if lockPhase == "free" then
                pcall(function()
                    R_RequestLock:FireServer()
                end)
                local t = tick()
                repeat
                    task.wait(0.2)
                until lockPhase ~= "free" or (tick() - t) > 3
            elseif lockPhase == "locked" then
                task.wait(1)
                local elapsed = tick() - phaseEnteredAt
                if elapsed >= math.max(lockSecondsLeft, 1) + 1 then
                    pcall(function()
                        R_RequestLock:FireServer()
                    end)
                    task.wait(0.5)
                end
            elseif lockPhase == "cooldown" then
                task.wait(1)
                local elapsed = tick() - phaseEnteredAt
                if elapsed >= math.max(lockSecondsLeft, 1) + 0.5 then
                    pcall(function()
                        R_RequestLock:FireServer()
                    end)
                    local t = tick()
                    repeat
                        task.wait(0.2)
                    until lockPhase ~= "cooldown" or (tick() - t) > 3
                end
            else
                task.wait(2)
            end
        end
    end

    local function 启动自动锁定()
        if State.自动锁定线程 then
            task.cancel(State.自动锁定线程)
        end
        State.自动锁定启用 = true
        State.自动锁定线程 = task.spawn(自动锁定循环)
    end

    local function 停止自动锁定()
        State.自动锁定启用 = false
        if State.自动锁定线程 then
            task.cancel(State.自动锁定线程)
            State.自动锁定线程 = nil
        end
    end

    local UPGRADE_INTERVAL = 0.5

    local function 启动自动升级(key)
        if State.自动升级线程[key] then
            task.cancel(State.自动升级线程[key])
        end
        State.自动升级[key] = true
        State.自动升级线程[key] = task.spawn(function()
            while State.自动升级[key] do
                pcall(function()
                    R_PurchaseUpgrade:FireServer(key)
                end)
                task.wait(UPGRADE_INTERVAL)
            end
        end)
    end

    local function 停止自动升级(key)
        State.自动升级[key] = false
        if State.自动升级线程[key] then
            task.cancel(State.自动升级线程[key])
            State.自动升级线程[key] = nil
        end
    end

    local function 停止所有自动升级()
        for _, key in ipairs({"等级", "最大", "锁定基地"}) do
            停止自动升级(key)
        end
    end

    local LEAVE_SCAN_INTERVAL = 0.5
    local SCRIPT_URL = "https://raw.githubusercontent.com/Raynisis1612/Merge-a-Nuke-Utilities/refs/heads/main/manu.lua"

    local function 获取我的岛屿地板()
        local bases = Workspace:FindFirstChild("Bases")
        if not bases then
            return nil
        end
        for _, base in ipairs(bases:GetChildren()) do
            local nukes = base:FindFirstChild("Nukes")
            if nukes then
                for _, nuke in ipairs(nukes:GetChildren()) do
                    if nuke:IsA("BasePart") and nuke:GetAttribute("OwnerUserId") == LocalPlayer.UserId then
                        return base:FindFirstChild("Floor")
                    end
                end
            end
        end
        return nil
    end

    local queueteleport = queue_on_teleport or (syn and syn.queue_on_teleport) or (fluxus and fluxus.queue_on_teleport)

    local function 执行重连()
        local rejoinDelay = State.重连延迟
        for i = math.ceil(rejoinDelay), 1, -1 do
            pcall(function()
                WindUI:Notify({
                    Title = "自动重连",
                    Content = i .. "秒后重连...",
                    Icon = "clock",
                    Duration = 1.1,
                })
            end)
            task.wait(1)
        end
        if SCRIPT_URL and SCRIPT_URL ~= "" then
            if queueteleport then
                local reexecCode = string.format('loadstring(game:HttpGet(%q))()', SCRIPT_URL)
                local ok, err = pcall(queueteleport, reexecCode)
                if ok then
                    print("[自动离开] 已排队重新执行脚本: " .. SCRIPT_URL)
                else
                    warn("[自动离开] queueteleport 调用失败: " .. err)
                end
            else
                warn("[自动离开] 未找到 queue_on_teleport — 重连后脚本不会自动重新执行")
            end
        end
        pcall(function()
            TeleportService:Teleport(game.PlaceId, LocalPlayer)
        end)
    end

    local function 自动离开循环()

    local function 格式化核弹价值(v)
        if v >= 1e15 then
            return string.format("%.2fQd", v / 1e15)
        elseif v >= 1e12 then
            return string.format("%.2fT", v / 1e12)
        elseif v >= 1e9 then
            return string.format("%.2fB", v / 1e9)
        elseif v >= 1e6 then
            return string.format("%.2fM", v / 1e6)
        elseif v >= 1e3 then
            return string.format("%.2fK", v / 1e3)
        else
            return tostring(math.floor(v))
        end
    end

    local function 扫描所有岛屿价值()
        local results = {}
        local bases = Workspace:FindFirstChild("Bases")
        if not bases then
            return results
        end
        local playerMap = {}
        for _, p in ipairs(Players:GetPlayers()) do
            playerMap[tostring(p.UserId)] = p.Name
        end
        for _, base in ipairs(bases:GetChildren()) do
            local nukesFolder = base:FindFirstChild("Nukes")
            if not nukesFolder then
                goto continue
            end
            local ownerMap = {}
            for _, nuke in ipairs(nukesFolder:GetChildren()) do
                if nuke:IsA("BasePart") then
                    local rawUid = nuke:GetAttribute("OwnerUserId")
                    local rawTier = nuke:GetAttribute("Tier")
                    if rawUid ~= nil and rawTier ~= nil then
                        local uidStr = tostring(rawUid)
                        local tierNum = tonumber(rawTier)
                        if tierNum and playerMap[uidStr] then
                            local val = 2 ^ (tierNum - 1)
                            if not ownerMap[uidStr] then
                                ownerMap[uidStr] = {
                                    userId = rawUid,
                                    name = playerMap[uidStr],
                                    totalValue = 0,
                                    floor = base:FindFirstChild("Floor"),
                                }
                            end
                            ownerMap[uidStr].totalValue = ownerMap[uidStr].totalValue + val
                        end
                    end
                end
            end
            for _, entry in pairs(ownerMap) do
                local found = false
                for _, r in ipairs(results) do
                    if r.userId == entry.userId then
                        r.totalValue = r.totalValue + entry.totalValue
                        found = true
                        break
                    end
                end
                if not found then
                    table.insert(results, entry)
                end
            end
            ::continue::
        end
        table.sort(results, function(a, b)
            return a.totalValue > b.totalValue
        end)
        return results
    end

    local function 等级转数值(等级)
        if 等级 <= 52 then
            return tostring(math.floor(2 ^ (等级 - 1)))
        else
            local exp = math.floor((等级 - 1) * math.log10(2))
            local m = math.floor((10 ^ ((等级 - 1) * math.log10(2) - exp)) * 10) / 10
            return m .. "e+" .. exp
        end

    local function 设置段落描述(para, text)
        if not para then return end
        if para.setDesc then
            pcall(function() para:setDesc(text) end)
        elseif para.SetDesc then
            pcall(function() para:SetDesc(text) end)
        elseif para.Set then
            pcall(function() para:Set({ Desc = text }) end)
        end
    end
    local isOverlayEnabled = false
    task.spawn(function()
        while true do
            local ok, err = pcall(function()
                local scan = 扫描所有岛屿价值()
                local text
                if #scan == 0 then
                    text = "未找到活跃岛屿。"
                else
                    local lines = {}
                    for i, entry in ipairs(scan) do
                        local tag = tostring(entry.userId) == tostring(LocalPlayer.UserId) and " ◀ 你" or ""
                        table.insert(lines, string.format("%d. %s — %s%s", i, entry.name, 格式化核弹价值(entry.totalValue), tag))
                    end
                    text = table.concat(lines, "\n")
                end
                设置段落描述(vtPara, text)
                更新价值追踪器显示(scan, isOverlayEnabled)
            end)
            if not ok then
                设置段落描述(vtPara, "错误: " .. tostring(err))
            end
            task.wait(1.5)
        end
    end)
