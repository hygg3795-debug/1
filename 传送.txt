return function(Window, WindUI)
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer

    local tab = Window:Tab({Title = "传送", Icon = "map-pin"})

    local function teleportTo(position)
        local char = LocalPlayer.Character
        if not char then
            WindUI:Notify({Title = "传送失败", Content = "角色未加载", Duration = 2, Icon = "x"})
            return
        end
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then
            root.CFrame = CFrame.new(position)
            WindUI:Notify({Title = "传送完成", Content = string.format("已传送到 (%.1f, %.1f, %.1f)", position.X, position.Y, position.Z), Duration = 2, Icon = "check"})
        else
            WindUI:Notify({Title = "传送失败", Content = "未找到HumanoidRootPart", Duration = 2, Icon = "x"})
        end
    end

    tab:Button({
        Title = "传送平台",
        Icon = "map-pin",
        Callback = function()
            teleportTo(Vector3.new(-273, 179.5, 394))
        end,
    })

    tab:Button({
        Title = "传送小岛",
        Icon = "map-pin",
        Callback = function()
            teleportTo(Vector3.new(-115.6, 48.6, 6.9))
        end,
    })

    tab:Button({
        Title = "传送彩蛋",
        Icon = "star",
        Callback = function()
            teleportTo(Vector3.new(-5059.2, 264.0, 6078.0))
        end,
    })

    tab:Button({
        Title = "传送虚空",
        Icon = "skull",
        Callback = function()
            teleportTo(Vector3.new(0, -100000, 0))
        end,
    })

    return tab
end
