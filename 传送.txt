return function(Window, WindUI)
    local LocalPlayer = game.Players.LocalPlayer
    local RunService = game:GetService("RunService")
    
    local function parseCoords(text)
        if not text or text == "" then return nil, nil, nil end
        local nums = {}
        for num in string.gmatch(text, "[-]?%d+%.?%d*") do
            table.insert(nums, tonumber(num))
        end
        if #nums >= 3 then
            return nums[1], nums[2], nums[3]
        end
        return nil, nil, nil
    end
    
    local function teleportTo(position)
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = CFrame.new(position)
        end
    end
    
    local TeleportTab = Window:CreateTab({
        Title = "传送",
        Icon = "move"
    })
    
    local coordLabel = TeleportTab:CreateLabel({
        Title = "当前坐标",
        Description = "X: 0.00  Y: 0.00  Z: 0.00"
    })
    
    local function updateCoordDisplay()
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local pos = char.HumanoidRootPart.Position
            coordLabel:SetDescription(string.format("X: %.2f  Y: %.2f  Z: %.2f", pos.X, pos.Y, pos.Z))
        end
    end
    
    updateCoordDisplay()
    task.spawn(function()
        while true do
            task.wait(0.5)
            updateCoordDisplay()
        end
    end)
    
    local inputBox = TeleportTab:CreateInput({
        Title = "目标坐标",
        Placeholder = "输入坐标 (例如: 100,200,300)"
    })
    
    TeleportTab:CreateButton({
        Title = "复制坐标",
        Icon = "copy",
        Callback = function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local pos = char.HumanoidRootPart.Position
                local coordText = string.format("%.2f,%.2f,%.2f", pos.X, pos.Y, pos.Z)
                setclipboard(coordText)
                WindUI:Notify({
                    Title = "已复制",
                    Content = coordText,
                    Duration = 2,
                    Icon = "copy"
                })
            end
        end
    })
    
    TeleportTab:CreateButton({
        Title = "传送",
        Icon = "send",
        Callback = function()
            local input = inputBox:GetText()
            if input == "" or input:match("^%s*$") then
                WindUI:Notify({
                    Title = "错误",
                    Content = "请输入坐标",
                    Duration = 2,
                    Icon = "alert"
                })
                return
            end
            local cleaned = input:gsub("，", ","):gsub("　", " ")
            local x, y, z = parseCoords(cleaned)
            if not x or not y or not z then
                WindUI:Notify({
                    Title = "错误",
                    Content = "坐标格式错误，请使用 x,y,z 格式",
                    Duration = 2,
                    Icon = "alert"
                })
                return
            end
            teleportTo(Vector3.new(x, y, z))
            WindUI:Notify({
                Title = "传送成功",
                Content = string.format("已传送到 (%.2f, %.2f, %.2f)", x, y, z),
                Duration = 2,
                Icon = "check"
            })
        end
    })
end
