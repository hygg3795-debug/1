return function(Window, WindUI)
    local player = game.Players.LocalPlayer
    local replicatedStorage = game:GetService("ReplicatedStorage")

    local autoSellActive = false
    local removeGrassActive = false

    local tab = Window:Tab({Title = "+1割草冒险", Icon = ""})

    tab:Toggle({
        Title = "自动出售",
        Value = false,
        Callback = function(state)
            autoSellActive = state
            if state then
                task.spawn(function()
                    while autoSellActive do
                        task.wait(0.2)
                        for _, container in ipairs({replicatedStorage, player:FindFirstChild("PlayerGui"), workspace}) do
                            if container then
                                for _, remote in ipairs(container:GetDescendants()) do
                                    if remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction") then
                                        local rname = string.lower(remote.Name)
                                        if string.find(rname, "sell") or string.find(rname, "sellall") or string.find(rname, "harvest") then
                                            pcall(function()
                                                if remote:IsA("RemoteEvent") then
                                                    remote:FireServer("All")
                                                    remote:FireServer(true)
                                                    remote:FireServer()
                                                else
                                                    remote:InvokeServer("All")
                                                end
                                            end)
                                        end
                                    end
                                end
                            end
                        end
                    end
                end)
            end
        end
    })

    tab:Toggle({
        Title = "清除草地",
        Value = false,
        Callback = function(state)
            removeGrassActive = state
            if state then
                task.spawn(function()
                    while removeGrassActive do
                        task.wait(0.1)
                        local char = player.Character
                        local root = char and char:FindFirstChild("HumanoidRootPart")
                        for _, obj in ipairs(workspace:GetDescendants()) do
                            if obj:IsA("BasePart") then
                                local pname = string.lower(obj.Name)
                                local parentName = string.lower(obj.Parent.Name)
                                if string.find(pname, "grass") or string.find(parentName, "grass") then
                                    pcall(function()
                                        if obj:FindFirstChildOfClass("TouchTransmitter") and root and firetouchinterest then
                                            firetouchinterest(obj, root, 0)
                                            task.wait()
                                            firetouchinterest(obj, root, 1)
                                        end
                                        obj:Destroy()
                                    end)
                                end
                            end
                        end
                        for _, container in ipairs({replicatedStorage, player:FindFirstChild("PlayerGui"), workspace}) do
                            if container then
                                for _, item in ipairs(container:GetDescendants()) do
                                    if item:IsA("RemoteEvent") or item:IsA("RemoteFunction") then
                                        local lname = string.lower(item.Name)
                                        if string.find(lname, "cut") or string.find(lname, "swing") or string.find(lname, "grass") or string.find(lname, "remove") then
                                            pcall(function()
                                                if item:IsA("RemoteEvent") then
                                                    item:FireServer()
                                                    item:FireServer(true)
                                                else
                                                    item:InvokeServer()
                                                end
                                            end)
                                        end
                                    end
                                end
                            end
                        end
                    end
                end)
            end
        end
    })
end
