return function(Window, WindUI)
    local tab = Window:Tab({Title = "信息", Icon = "info"})

    local Players = game:GetService("Players")
    local plr = Players.LocalPlayer
    if not plr then
        repeat task.wait() until Players.LocalPlayer
        plr = Players.LocalPlayer
    end

    local function getExecutor()
        if identifyexecutor then return identifyexecutor() end
        if getexecutorname then return getexecutorname() end
        return "未知/通用"
    end

    local function getResolution()
        local ok, res = pcall(function() return game:GetService("GuiService"):GetScreenResolution() end)
        if ok and res then return res.X .. "x" .. res.Y end
        return "无法获取"
    end

    local username = plr.Name or "无法获取"
    local displayName = plr.DisplayName or "无法获取"
    local userId = tostring(plr.UserId or 0)
    local locale = tostring(plr.LocaleId or "未知")
    local accountAge = (plr.AccountAge or 0) .. " 天 (" .. string.format("%.2f", (plr.AccountAge or 0) / 365) .. " 年)"
    local executorName = getExecutor()
    local placeIdStr = tostring(game.PlaceId)
    local placeVerStr = tostring(game.PlaceVersion)
    local jobIdStr = (game.JobId ~= "" and game.JobId or "单人/本地服务器")
    local maxPlrsStr = tostring(Players.MaxPlayers)
    local rbxVersion = version()
    local deviceType = (game:GetService("UserInputService").TouchEnabled and "移动设备" or "电脑")
    local resolutionStr = getResolution()

    local userSection = tab:Section({ Title = "玩家与账号", Box = true })
    userSection:Paragraph({ Title = "用户名", Desc = username, Content = username })
    userSection:Paragraph({ Title = "显示名称", Desc = displayName, Content = displayName })
    userSection:Paragraph({ Title = "用户 ID", Desc = userId, Content = userId })
    userSection:Paragraph({ Title = "语言", Desc = locale, Content = locale })
    local pCountry = userSection:Paragraph({ Title = "国家/地区", Desc = "正在获取...", Content = "正在获取..." })
    userSection:Paragraph({ Title = "账号年龄", Desc = accountAge, Content = accountAge })

    local envSection = tab:Section({ Title = "游戏与环境", Box = true })
    local pGameName = envSection:Paragraph({ Title = "游戏名称", Desc = "正在获取...", Content = "正在获取..." })
    envSection:Paragraph({ Title = "游戏 ID (PlaceId)", Desc = placeIdStr, Content = placeIdStr })
    envSection:Paragraph({ Title = "游戏版本", Desc = placeVerStr, Content = placeVerStr })
    envSection:Paragraph({ Title = "服务器 ID (JobId)", Desc = jobIdStr, Content = jobIdStr })
    envSection:Paragraph({ Title = "最大玩家数", Desc = maxPlrsStr, Content = maxPlrsStr })
    envSection:Paragraph({ Title = "注入器", Desc = executorName, Content = executorName })
    envSection:Paragraph({ Title = "Roblox 版本", Desc = rbxVersion, Content = rbxVersion })
    envSection:Paragraph({ Title = "设备类型", Desc = deviceType, Content = deviceType })
    envSection:Paragraph({ Title = "屏幕分辨率", Desc = resolutionStr, Content = resolutionStr })

    local char = plr.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local maxHealth = hum and tostring(hum.MaxHealth) or (char and "无法获取" or "角色未加载")
    local jumpPower = hum and tostring(hum.JumpPower) or (char and "无法获取" or "角色未加载")
    local walkSpeed = hum and tostring(hum.WalkSpeed) or (char and "无法获取" or "角色未加载")

    local statsSection = tab:Section({ Title = "角色数据", Box = true })
    statsSection:Paragraph({ Title = "最大生命值", Desc = maxHealth, Content = maxHealth })
    statsSection:Paragraph({ Title = "最大跳跃力", Desc = jumpPower, Content = jumpPower })
    statsSection:Paragraph({ Title = "最大移动速度", Desc = walkSpeed, Content = walkSpeed })

    task.spawn(function()
        local country = "无法获取"
        pcall(function()
            country = game:GetService("LocalizationService"):GetCountryRegionForPlayerAsync(plr)
        end)
        pcall(function()
            if pCountry.SetDesc then pCountry:SetDesc(country) end
            if pCountry.SetContent then pCountry:SetContent(country) end
        end)

        local gameName = "无法获取"
        pcall(function()
            gameName = game:GetService("MarketplaceService"):GetProductInfo(game.PlaceId).Name
        end)
        pcall(function()
            if pGameName.SetDesc then pGameName:SetDesc(gameName) end
            if pGameName.SetContent then pGameName:SetContent(gameName) end
        end)
    end)

    return tab
end
