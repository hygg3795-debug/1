return function(Window, WindUI)
    local success, err = pcall(function()
        print("黑洞模块开始执行")
        
        local blackholeTab = nil
        local createMethods = {"Tab", "AddTab", "CreateTab", "NewTab"}
        
        for _, method in ipairs(createMethods) do
            if Window[method] then
                blackholeTab = Window[method](Window, {Title = "黑洞", Icon = "circle"})
                if blackholeTab then 
                    print("黑洞Tab创建成功，使用方式: " .. method)
                    break 
                end
            end
        end
        
        if not blackholeTab then
            error("无法创建黑洞Tab，请检查WindUI版本")
        end

        local Players = game:GetService("Players")
        local RunService = game:GetService("RunService")
        local Workspace = game:GetService("Workspace")
        local LocalPlayer = Players.LocalPlayer

        local section = blackholeTab:Section({Title = "黑洞控制", Box = true})

        local parts = {}
        local enabled = false
        local con = nil
        local partCon = nil
        local removeCon = nil

        local config = {
            radius = 50,
            height = 100,
            rotationSpeed = 10,
            attractionStrength = 1000
        }

        local function retainPart(part)
            if part:IsA("BasePart") and not part.Anchored and part:IsDescendantOf(Workspace) then
                if part.Parent == LocalPlayer.Character or part:IsDescendantOf(LocalPlayer.Character) then return false end
                pcall(function()
                    part.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0, 0, 0)
                    part.CanCollide = false
                end)
                return true
            end
            return false
        end

        local function addPart(part)
            if retainPart(part) and not table.find(parts, part) then
                table.insert(parts, part)
            end
        end

        local function removePart(part)
            local idx = table.find(parts, part)
            if idx then table.remove(parts, idx) end
        end

        local function refreshParts()
            parts = {}
            for _, part in pairs(Workspace:GetDescendants()) do
                addPart(part)
            end
        end

        local function startBlackhole()
            if con then return end
            enabled = true
            refreshParts()
            
            if partCon then partCon:Disconnect() end
            partCon = Workspace.DescendantAdded:Connect(addPart)
            
            if removeCon then removeCon:Disconnect() end
            removeCon = Workspace.DescendantRemoving:Connect(removePart)

            con = RunService.Heartbeat:Connect(function()
                if not enabled then return end
                local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if not root then return end
                local center = root.Position
                for _, part in pairs(parts) do
                    if part.Parent and not part.Anchored then
                        local pos = part.Position
                        local dist = (Vector3.new(pos.X, center.Y, pos.Z) - center).Magnitude
                        local angle = math.atan2(pos.Z - center.Z, pos.X - center.X)
                        local newAngle = angle + math.rad(config.rotationSpeed)
                        local targetPos = Vector3.new(
                            center.X + math.cos(newAngle) * math.min(config.radius, dist),
                            center.Y + (config.height * (math.abs(math.sin((pos.Y - center.Y) / config.height)))),
                            center.Z + math.sin(newAngle) * math.min(config.radius, dist)
                        )
                        local dir = (targetPos - part.Position).unit
                        part.Velocity = dir * config.attractionStrength
                    end
                end
            end)
        end

        local function stopBlackhole()
            enabled = false
            if con then con:Disconnect() con = nil end
            if partCon then partCon:Disconnect() partCon = nil end
            if removeCon then removeCon:Disconnect() removeCon = nil end
            parts = {}
        end

        local toggleControl = section:Toggle({
            Title = "开启黑洞",
            Value = false,
            Callback = function(val)
                if val then
                    startBlackhole()
                    WindUI:Notify({Title = "黑洞", Content = "已开启", Duration = 2, Icon = "check"})
                else
                    stopBlackhole()
                    WindUI:Notify({Title = "黑洞", Content = "已关闭", Duration = 2, Icon = "x"})
                end
            end,
        })

        section:Slider({
            Title = "半径",
            Value = 50,
            Min = 0,
            Max = 500,
            Callback = function(val)
                config.radius = val
            end,
        })

        section:Slider({
            Title = "高度",
            Value = 100,
            Min = 0,
            Max = 500,
            Callback = function(val)
                config.height = val
            end,
        })

        section:Slider({
            Title = "转速",
            Value = 10,
            Min = 0,
            Max = 200,
            Callback = function(val)
                config.rotationSpeed = val
            end,
        })

        section:Slider({
            Title = "吸力",
            Value = 1000,
            Min = 0,
            Max = 50000,
            Callback = function(val)
                config.attractionStrength = val
            end,
        })

        print("黑洞模块加载完成 ✅")
        return blackholeTab
    end)
    
    if not success then
        WindUI:Notify({Title = "黑洞模块错误", Content = tostring(err), Duration = 3, Icon = "alert"})
        print("黑洞模块错误:", err)
    end
end
