    local function createGameSection(title)
        local section = tab:Section({
            Title = title,
            Collapsible = true,
            Box = true,
            DefaultOpen = true
        })
        return section
    end

    local connections = {}
    local activeStates = {}
    local yPositions = {}
    local dirs = {}

    local function stopGame(gameName)
        if connections[gameName] then
            connections[gameName]:Disconnect()
            connections[gameName] = nil
        end
        activeStates[gameName] = false
    end

    local function startSingle(gameName, config)
        local key = gameName
        if connections[key] then return end
        activeStates[key] = true
        if yPositions[key] == nil then
            yPositions[key] = config.baseY
        end
        if dirs[key] == nil then
            dirs[key] = 1
        end
        connections[key] = RunService.Heartbeat:Connect(function()
            if not activeStates[key] then return end
            local char = LocalPlayer.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                yPositions[key] = yPositions[key] + config.speed * dirs[key]
                if yPositions[key] <= config.baseY + config.minOffset then
                    dirs[key] = 1
                    yPositions[key] = config.baseY + config.minOffset
                end
                if yPositions[key] >= config.baseY + config.maxOffset then
                    dirs[key] = -1
                    yPositions[key] = config.baseY + config.maxOffset
                end
                hrp.CFrame = CFrame.new(config.pos) + Vector3.new(0, yPositions[key] - config.baseY, 0)
            end
        end)
    end

    local function startMulti(gameName, config, posIndex)
        local key = gameName .. "_" .. tostring(posIndex)
        if connections[key] then return end
        activeStates[key] = true
        local pos = config.positions[posIndex]
        if yPositions[key] == nil then
            yPositions[key] = config.baseY
        end
        if dirs[key] == nil then
            dirs[key] = 1
        end
        connections[key] = RunService.Heartbeat:Connect(function()
            if not activeStates[key] then return end
            local char = LocalPlayer.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                yPositions[key] = yPositions[key] + config.speed * dirs[key]
                if yPositions[key] <= config.baseY + config.minOffset then
                    dirs[key] = 1
                    yPositions[key] = config.baseY + config.minOffset
                end
                if yPositions[key] >= config.baseY + config.maxOffset then
                    dirs[key] = -1
                    yPositions[key] = config.baseY + config.maxOffset
                end
                hrp.CFrame = CFrame.new(pos) + Vector3.new(0, yPositions[key] - config.baseY, 0)
            end
        end)
    end

    local function createGameToggle(gameName)
        local config = gameConfigs[gameName]
        if not config then return end

        local section = createGameSection(gameName)

        if config.type == "multi" then
            for i = 1, #config.positions do
                section:Toggle({
                    Title = "自动胜利世界" .. i,
                    Value = false,
                    Callback = function(val)
                        if val then
                            startMulti(gameName, config, i)
                        else
                            local key = gameName .. "_" .. tostring(i)
                            if connections[key] then
                                connections[key]:Disconnect()
                                connections[key] = nil
                            end
                            activeStates[key] = false
                        end
                    end
                })
            end
        else
            section:Toggle({
                Title = "自动胜利",
                Value = false,
                Callback = function(val)
                    if val then
                        startSingle(gameName, config)
                    else
                        stopGame(gameName)
                    end
                end
            })
        end
    end

    for name, _ in pairs(gameConfigs) do
        createGameToggle(name)
    end

    return tab
