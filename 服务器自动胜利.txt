return function(Window, WindUI)
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local LocalPlayer = Players.LocalPlayer

    local tab = Window:Tab({Title = "服务器自动胜利", Icon = "trophy"})

    local gameConfigs = {
        ["+1速度猴子逃脱"] = {
            type = "generic",
            pos1 = Vector3.new(-9460.94, 389.29, -256.31),
            pos2 = Vector3.new(-3604.62, 154.21, -9378.90),
            baseY = 389.29,
            minOffset = -3,
            maxOffset = 4,
            speed = 0.5
        },
        ["阿拉伯键盘"] = {
            type = "generic",
            pos = Vector3.new(-7344.6, 1877.7, 1379.7),
            baseY = 1877.7,
            minOffset = -3,
            maxOffset = 4,
            speed = 0.5
        },
        ["超高速跑者"] = {
            type = "generic",
            pos = Vector3.new(0, 6.5, -9076),
            baseY = 1,
            minOffset = -3,
            maxOffset = 4,
            speed = 0.5
        },
        ["每步+1速度"] = {
            type = "generic",
            pos = Vector3.new(8294.85, 201.73, -7598.20),
            baseY = 1,
            minOffset = -3,
            maxOffset = 4,
            speed = 10
        },
        ["+1斧头挥动逃脱"] = {
            type = "generic",
            pos = Vector3.new(1769.53, 6.11, 1222.30),
            baseY = 1,
            minOffset = -3,
            maxOffset = 4,
            speed = 0.5
        },
        ["+1速度键盘逃脱水下"] = {
            type = "underwater",
            pos = Vector3.new(-14024.0, 752.7, 3576.2),
            baseY = 0,
            minOffset = -2,
            maxOffset = 2,
            speed = 5
        }
    }

    local function createGameSection(title)
        local section = tab:Section({
            Title = title,
            Collapsible = true,
            Box = true,
            DefaultOpen = true
        })
        return section
    end

    local connections = {}
    local activeStates = {}
    local timers = {}
    local yPositions = {}
    local dirs = {}

    local function stopGame(gameName, subKey)
        local key = gameName .. "_" .. (subKey or "default")
        if connections[key] then
            connections[key]:Disconnect()
            connections[key] = nil
        end
        activeStates[key] = false
        timers[key] = 0
    end

    local function startGeneric(gameName, config, subKey)
        local key = gameName .. "_" .. subKey
        if connections[key] then return end
        activeStates[key] = true
        if yPositions[key] == nil then
            yPositions[key] = config.baseY
        end
        if dirs[key] == nil then
            dirs[key] = 1
        end
        connections[key] = RunService.Heartbeat:Connect(function()
            if not activeStates[key] then return end
            local char = LocalPlayer.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                yPositions[key] = yPositions[key] + config.speed * dirs[key]
                if yPositions[key] <= config.baseY + config.minOffset then
                    dirs[key] = 1
                    yPositions[key] = config.baseY + config.minOffset
                end
                if yPositions[key] >= config.baseY + config.maxOffset then
                    dirs[key] = -1
                    yPositions[key] = config.baseY + config.maxOffset
                end
                hrp.CFrame = CFrame.new(config.pos) + Vector3.new(0, yPositions[key] - config.baseY, 0)
            end
        end)
    end

    local function startUnderwater(gameName, config, subKey)
        local key = gameName .. "_" .. subKey
        if connections[key] then return end
        activeStates[key] = true
        if yPositions[key] == nil then
            yPositions[key] = config.baseY
        end
        if dirs[key] == nil then
            dirs[key] = 1
        end
        connections[key] = RunService.Heartbeat:Connect(function()
            if not activeStates[key] then return end
            local char = LocalPlayer.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                yPositions[key] = yPositions[key] + config.speed * dirs[key]
                if yPositions[key] <= config.baseY + config.minOffset then
                    dirs[key] = 1
                    yPositions[key] = config.baseY + config.minOffset
                end
                if yPositions[key] >= config.baseY + config.maxOffset then
                    dirs[key] = -1
                    yPositions[key] = config.baseY + config.maxOffset
                end
                hrp.CFrame = CFrame.new(config.pos) + Vector3.new(0, yPositions[key], 0)
            end
        end)
    end

    local function createGameToggle(gameName)
        local config = gameConfigs[gameName]
        if not config then return end

        local section = createGameSection(gameName)

        if config.type == "monkey" then
            section:Toggle({
                Title = "自动胜利世界1",
                Value = false,
                Callback = function(val)
                    if val then
                        startMonkey(gameName, config, config.pos1, "pos1")
                    else
                        stopGame(gameName, "pos1")
                    end
                end
            })
            section:Toggle({
                Title = "自动胜利世界2",
                Value = false,
                Callback = function(val)
                    if val then
                        startMonkey(gameName, config, config.pos2, "pos2")
                    else
                        stopGame(gameName, "pos2")
                    end
                end
            })
        elseif config.type == "underwater" then
            section:Toggle({
                Title = "自动胜利",
                Value = false,
                Callback = function(val)
                    if val then
                        startUnderwater(gameName, config, "default")
                    else
                        stopGame(gameName, "default")
                    end
                end
            })
        else
            section:Toggle({
                Title = "自动胜利",
                Value = false,
                Callback = function(val)
                    if val then
                        startGeneric(gameName, config, "default")
                    else
                        stopGame(gameName, "default")
                    end
                end
            })
        end
    end

    for name, _ in pairs(gameConfigs) do
        createGameToggle(name)
    end

    return tab
end
