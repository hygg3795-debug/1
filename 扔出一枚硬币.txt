return function(Window, WindUI)
    local GameName = "扔出一枚硬币"
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer
    local UserInputService = game:GetService("UserInputService")
    local CoreGui = game:GetService("CoreGui")

    _G.AutoThrowActive = false
    _G.AutoSellActive = false
    _G.AutoBuyActive = false
    _G.AutoUpgradeLuck = false
    _G.AutoUpgradeValue = false
    _G.ThrowMultiplier = 3
    _G.ThrowRemote = nil
    _G.ThrowArgs = {}
    _G.SellRemote = nil
    _G.SellArgs = {}
    _G.UpgradeRemote = nil
    _G.UpgradeArgs = {}

    local function getCoinName()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        if not pg then return nil end
        local coinLabel = pg:FindFirstChild("UiFolder")
            and pg.UiFolder:FindFirstChild("Main")
            and pg.UiFolder.Main:FindFirstChild("HUD")
            and pg.UiFolder.Main.HUD:FindFirstChild("Coin")
            and pg.UiFolder.Main.HUD.Coin:FindFirstChild("Main")
            and pg.UiFolder.Main.HUD.Coin.Main:FindFirstChild("CoinName")
        if coinLabel and coinLabel:IsA("TextLabel") and coinLabel.Text ~= "" then
            return coinLabel.Text
        end
        return nil
    end

    local function getMultiplier()
        local pg = LocalPlayer:FindFirstChild("PlayerGui")
        if not pg then return 0 end
        local multBar = pg:FindFirstChild("UiFolder")
            and pg.UiFolder:FindFirstChild("Main")
            and pg.UiFolder.Main:FindFirstChild("HUD")
            and pg.UiFolder.Main.HUD:FindFirstChild("ThrowBar")
            and pg.UiFolder.Main.HUD.ThrowBar:FindFirstChild("CurrentMulti")
        if multBar and multBar:IsA("Frame") then
            return multBar.Size.X.Scale * 3
        end
        return 0
    end

    local function getHRP()
        local char = LocalPlayer.Character
        return char and char:FindFirstChild("HumanoidRootPart")
    end

    local coinList = {
        "Paradox Coin", "Lucky Coin", "Golden Coin", "Obsidian Coin",
        "Platinum Coin", "Ruby Coin", "Emerald Coin", "Amethyst Coin",
        "Topaz Coin", "Diamond Coin", "Staff Token", "VIP Token",
        "Developer Token", "Diamond Token"
    }

    local function scanRemotes()
        for _, item in ipairs(ReplicatedStorage:GetDescendants()) do
            if item:IsA("RemoteEvent") or item:IsA("RemoteFunction") then
                local name = string.lower(item.Name)
                if string.find(name, "throw") or string.find(name, "toss") or string.find(name, "coin") then
                    if not _G.ThrowRemote then
                        _G.ThrowRemote = item
                        _G.ThrowArgs = {}
                    end
                elseif string.find(name, "sell") then
                    if not _G.SellRemote then
                        _G.SellRemote = item
                        _G.SellArgs = {}
                    end
                elseif string.find(name, "upgrade") or string.find(name, "buy") then
                    if not _G.UpgradeRemote then
                        _G.UpgradeRemote = item
                        _G.UpgradeArgs = {}
                    end
                end
            end
        end
    end
    pcall(scanRemotes)

    local hookSuccess, hookError = pcall(function()
        local oldNamecall
        oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
            local method = getnamecallmethod()
            local args = {...}
            if method == "FireServer" or method == "InvokeServer" then
                local name = string.lower(self.Name)
                if string.find(name, "throw") or string.find(name, "toss") or string.find(name, "coin") then
                    _G.ThrowRemote = self
                    _G.ThrowArgs = args
                elseif string.find(name, "sell") then
                    _G.SellRemote = self
                    _G.SellArgs = args
                elseif string.find(name, "upgrade") or string.find(name, "buy") then
                    _G.UpgradeRemote = self
                    _G.UpgradeArgs = args
                end
            end
            return oldNamecall(self, ...)
        end)
    end)

    local throwCooldown = 0
    task.spawn(function()
        while true do
            task.wait(0.2)
            if _G.AutoThrowActive and _G.ThrowRemote then
                local currentMult = getMultiplier()
                if currentMult >= _G.ThrowMultiplier then
                    local coinName = getCoinName()
                    if not coinName then
                        continue
                    end
                    local hrp = getHRP()
                    if not hrp then
                        continue
                    end
                    local pos = hrp.Position + Vector3.new(0, -0.5, -2)
                    local mult = math.floor(_G.ThrowMultiplier)
                    if _G.ThrowMultiplier == mult then
                        pcall(function()
                            _G.ThrowRemote:FireServer(mult, pos, coinName, nil, nil)
                        end)
                    else
                        pcall(function()
                            _G.ThrowRemote:FireServer(_G.ThrowMultiplier, pos, coinName, nil, nil)
                        end)
                    end
                    throwCooldown = tick() + 0.5
                end
            end
        end
    end)

    local sellCooldown = 0
    task.spawn(function()
        while true do
            task.wait(0.5)
            if _G.AutoSellActive and _G.SellRemote then
                if tick() >= sellCooldown then
                    pcall(function()
                        _G.SellRemote:FireServer()
                    end)
                    sellCooldown = tick() + 2
                end
            end
        end
    end)

    local upgradeCooldown = 0
    task.spawn(function()
        while true do
            task.wait(1.0)
            if (not _G.AutoUpgradeLuck and not _G.AutoUpgradeValue) then
                continue
            end
            if tick() < upgradeCooldown then
                continue
            end
            if _G.UpgradeRemote then
                local args = _G.UpgradeArgs or {}
                if _G.AutoUpgradeLuck then
                    pcall(function()
                        if #args > 0 then
                            _G.UpgradeRemote:FireServer(unpack(args))
                        else
                            _G.UpgradeRemote:FireServer("Luck Multiplier")
                        end
                    end)
                end
                if _G.AutoUpgradeValue then
                    pcall(function()
                        if #args > 0 then
                            _G.UpgradeRemote:FireServer(unpack(args))
                        else
                            _G.UpgradeRemote:FireServer("Value Multiplier")
                        end
                    end)
                end
                upgradeCooldown = tick() + 1.5
            end
        end
    end)

    local buyCooldown = 0
    task.spawn(function()
        while true do
            task.wait(1.0)
            if not _G.AutoBuyActive then
                continue
            end
            if tick() < buyCooldown then
                continue
            end
            if not getCoinName() then
                local buyRemote = ReplicatedStorage:FindFirstChild("Assets")
                    and ReplicatedStorage.Assets:FindFirstChild("Events")
                    and ReplicatedStorage.Assets.Events:FindFirstChild("BuyCoin")
                if buyRemote then
                    for _, name in ipairs(coinList) do
                        pcall(function()
                            buyRemote:FireServer(name)
                        end)
                        task.wait(0.1)
                    end
                    buyCooldown = tick() + 3
                end
            end
        end
    end)

    local tab = Window:Tab({Title = GameName, Icon = "coins"})

    tab:Toggle({
        Title = "自动投币",
        Value = false,
        Callback = function(state)
            _G.AutoThrowActive = state
        end
    })
    tab:Slider({
        Title = "投币倍率",
        Step = 0.1,
        Value = { Min = 1, Max = 3, Default = 3 },
        Width = 200,
        IsTextbox = true,
        Callback = function(val)
            _G.ThrowMultiplier = val
        end
    })
    tab:Divider()

    tab:Toggle({
        Title = "自动出售",
        Value = false,
        Callback = function(state)
            _G.AutoSellActive = state
        end
    })
    tab:Divider()

    tab:Toggle({
        Title = "自动购买硬币",
        Value = false,
        Callback = function(state)
            _G.AutoBuyActive = state
        end
    })
    tab:Divider()

    local upgradeSection = tab:Section({
        Title = "自动升级",
        Collapsible = true,
        Box = true,
        DefaultOpen = false,
    })
    upgradeSection:Toggle({
        Title = "升级（运气）",
        Value = false,
        Callback = function(state)
            _G.AutoUpgradeLuck = state
        end
    })
    upgradeSection:Toggle({
        Title = "升级（价值）",
        Value = false,
        Callback = function(state)
            _G.AutoUpgradeValue = state
        end
    })

    return tab
end
