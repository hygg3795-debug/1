return function(Window, WindUI)
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local LocalPlayer = Players.LocalPlayer

    -- 创建"极速传奇"Tab
    local tab = Window:Tab({Title = "极速传奇", Icon = "crown"})

    -- ============================================================
    -- 列表1: 主要
    -- ============================================================
    local list1 = tab:List({
        Title = "主要",
        Icon = "house",
        Opened = false,
    })

    list1:Button({
        Title = "收集全部宝箱",
        Callback = function()
            for _, v in pairs(game.ReplicatedStorage.chestRewards:GetChildren()) do
                game.ReplicatedStorage.rEvents.checkChestRemote:InvokeServer(v.Name)
            end
            WindUI:Notify({
                Title = "提示",
                Content = "已收集全部宝箱",
                Duration = 3,
                Icon = "gift"
            })
        end
    })

    list1:Button({
        Title = "解锁全部通行证",
        Callback = function()
            for i, v in ipairs(game:GetService("ReplicatedStorage").gamepassIds:GetChildren()) do
                v.Parent = game.Players.LocalPlayer.ownedGamepasses
            end
            WindUI:Notify({
                Title = "提示",
                Content = "走几步即可",
                Duration = 3,
                Icon = "crown"
            })
        end
    })

    list1:Toggle({
        Title = "吸全部环",
        Default = false,
        Callback = function(state)
            getgenv().HoopEnabled = state
            if state then
                spawn(function()
                    while getgenv().HoopEnabled and task.wait() do
                        pcall(function()
                            local character = game.Players.LocalPlayer.Character
                            if character and character:FindFirstChild("HumanoidRootPart") then
                                local hrp = character.HumanoidRootPart
                                for _, hoop in ipairs(workspace.Hoops:GetChildren()) do
                                    if hoop.Name == "Hoop" then
                                        hoop.CFrame = hrp.CFrame
                                    end
                                end
                            end
                        end)
                    end
                end)
            end
        end
    })

    list1:Toggle({
        Title = "自动抽奖",
        Default = false,
        Callback = function(state)
            Interstellar.autoSpin = state
            if state then
                spawn(function()
                    while Interstellar.autoSpin do
                        pcall(function()
                            game:GetService("ReplicatedStorage").rEvents.openFortuneWheelRemote:InvokeServer("openFortuneWheel", game:GetService("ReplicatedStorage").fortuneWheelChances["Fortune Wheel"])
                        end)
                        task.wait(1)
                    end
                end)
            end
        end
    })

    list1:Toggle({
        Title = "自动收集奖励",
        Default = false,
        Callback = function(state)
            getgenv().AutoCollectReward = state
            if state then
                spawn(function()
                    while getgenv().AutoCollectReward do
                        pcall(function()
                            for i = 1, 8 do
                                local args = {
                                    "claimGift",
                                    i
                                }
                                game:GetService("ReplicatedStorage"):WaitForChild("rEvents"):WaitForChild("freeGiftClaimRemote"):InvokeServer(unpack(args))

                        end)
                        task.wait()
                    end
                end)
            end
        end

    -- ============================================================

        Title = "卡宠",
        Icon = "paw-print",
        Opened = false,
    })

        Title = "模式一",
        Opened = false,

        getorb = false

    subList1:Dropdown({
        Title = "选择地区",
        Values = areaDisplayNames,
        Value = "城市",
        Callback = function(Value)
            Interstellar.area = areaOptions[table.find(areaDisplayNames, Value)]

    })

    local function checkMaxLevel()
        local player = game.Players.LocalPlayer
        Interstellar.isMaxLevel = player:GetAttribute("IsMaxLevel") == true
    end

    RunService.Heartbeat:Connect(function()
        checkMaxLevel()
    end)

    subList1:Toggle({
        Title = "紫球",
        Default = false,
        Callback = function(state)
            Interstellar.getorb = state
            if state then
                spawn(function()
                    while Interstellar.getorb do
                        for i = 1, 20 do
                            game.ReplicatedStorage.rEvents.orbEvent:FireServer("collectOrb", "Ethereal Orb", Interstellar.area)
                        end
                        wait(0.03)
                    end
                end)
            end
        end
    })

    subList1:Toggle({
        Title = "红球",
        Default = false,
        Callback = function(state)
            Interstellar.getorb = state
            if state then
                spawn(function()
                    while Interstellar.getorb do
                        for i = 1, 20 do
                            game.ReplicatedStorage.rEvents.orbEvent:FireServer("collectOrb", "Red Orb", Interstellar.area)
                        end
                        wait(0.03)
                    end
                end)
            end
        end
    })

    subList1:Toggle({
        Title = "蓝球",
        Default = false,
        Callback = function(state)
            Interstellar.getorb = state
            if state then
                spawn(function()
                    while Interstellar.getorb do
                        for i = 1, 20 do
                            game.ReplicatedStorage.rEvents.orbEvent:FireServer("collectOrb", "Blue Orb", Interstellar.area)
                        end
                        wait(0.03)
                    end
                end)
            end
        end
    })

        Title = "橙球",
        Default = false,
        Callback = function(state)

            if state then

        Title = "经验球",
        Default = false,
        Callback = function(state)

            if state then

    -- 模式二
    local subList2 = list2:SubList({
        Title = "模式二",
        Opened = false,
    })

    local Interstellar2 = {
        area = "City",
        orbType = "Yellow Orb",
        rateEnabled = false,
        multiplier = 1,
        validAreas = {"City","Snow City","Magma City","Legends Highway","Speed Jungle","Desert","Space"},
        currentArea = "City",
        isMaxLevel = false
    }

    local function updateCurrentArea()
        Interstellar2.currentArea = game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:GetAttribute("CurrentArea") or "City"
    end

    local function checkMaxLevel2()
        local player = game.Players.LocalPlayer
        Interstellar2.isMaxLevel = player:GetAttribute("IsMaxLevel") == true
    end

    RunService.Heartbeat:Connect(function()
        updateCurrentArea()
        checkMaxLevel2()
    end)

    subList2:Dropdown({
        Title = "选择地区",
        Values = areaDisplayNames,
        Value = "城市",
        Callback = function(Value)
            local idx = table.find(areaDisplayNames, Value)
            if idx then
                Interstellar2.area = areaOptions[idx]
            else
                Interstellar2.area = "City"
            end
        end
    })

    local orbDisplayNames = {"紫球", "红球", "蓝球", "橙球", "经验球"}
    local orbTypes = {"Ethereal Orb", "Red Orb", "Blue Orb", "Orange Orb", "Yellow Orb"}

    subList2:Dropdown({
        Title = "选择球类型",
        Values = orbDisplayNames,
        Value = "",
        Callback = function(Value)
            local idx = table.find(orbDisplayNames, Value)
            if idx then
                Interstellar2.orbType = orbTypes[idx]
            else
                Interstellar2.orbType = ""
            end
        end
    })

    subList2:Input({
        Title = "输入倍速",
        Desc = "默认为1",
        Placeholder = "输入数值",
        Callback = function(value)
            Interstellar2.multiplier = math.clamp(tonumber(value) or 1, 1, 10000)
        end
    })

    subList2:Toggle({
        Title = "倍速卡宠总开关",
        Default = false,
        Callback = function(state)
            Interstellar2.rateEnabled = state
            if state then
                spawn(function()
                    local e = game:GetService("ReplicatedStorage").rEvents.orbEvent
                    while Interstellar2.rateEnabled do
                        if Interstellar2.isMaxLevel then
                            Interstellar2.rateEnabled = false
                            break
                        end
                        local total = 1 * Interstellar2.multiplier
                        for i = 1, total do
                            e:FireServer("collectOrb", Interstellar2.orbType, Interstellar2.area)
                        end
                        task.wait()
                    end
                end)
            end
        end
    })

    -- ============================================================
    -- 列表3: 宝石
    -- ============================================================
    local list3 = tab:List({
        Title = "宝石",
        Icon = "gem",
        Opened = false,
    })

    local gemAreas = {
        {name = "城市宝石", key = "sbsA", area = "City"},
        {name = "雪城宝石", key = "sbsB", area = "Snow City"},
        {name = "熔岩城宝石", key = "sbsC", area = "Magma City"},
        {name = "传奇公路宝石", key = "sbsD", area = "Legends Highway"},
        {name = "丛林宝石", key = "sbsE", area = "Speed Jungle"},
    }

    for _, gem in ipairs(gemAreas) do
        list3:Toggle({
            Title = gem.name,
            Default = false,
            Callback = function(state)
                getgenv()[gem.key] = state
                if state then
                    spawn(function()
                        while getgenv()[gem.key] do
                            for i = 1, 10 do
                                game:GetService("ReplicatedStorage").rEvents.orbEvent:FireServer("collectOrb", "Gem", gem.area)
                            end
                            wait()
                        end
                    end)
                end
            end
        })
    end

    -- ============================================================
    -- 列表4: 比赛
    -- ============================================================
    local list4 = tab:List({
        Title = "比赛",
        Icon = "flag",
        Opened = false,
    })

    local Maps = {}
    for _,v in pairs(game.workspace.raceMaps:GetChildren()) do
        table.insert(Maps, v.Name)
    end

    local selectedMap = ""

    list4:Toggle({
        Title = "自动比赛",
        Default = false,
        Callback = function(state)
            autoRace = state
            if state then
                spawn(function()
                    while autoRace do
                        game:GetService("ReplicatedStorage").rEvents.raceEvent:FireServer("joinRace")
                        wait()
                    end
                end)
            end
        end
    })

    list4:Dropdown({
        Title = "比赛终点传送",
        Values = Maps,
        Value = "",
        Callback = function(value)
            selectedMap = value
        end
    })

    list4:Button({
        Title = "点击传送",
        Callback = function()
            if selectedMap and game.workspace.raceMaps:FindFirstChild(selectedMap) then
                local char = game.Players.LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    char.HumanoidRootPart.CFrame = game.workspace.raceMaps[selectedMap].finishPart.CFrame
                end
            end
        end
    })

    list4:Toggle({
        Title = "自动胜利",
        Default = false,
        Callback = function(state)
            Interstellar.mainexe = state
            if Interstellar.mainexe then
                if game.PlaceId == 3101667897 then
                    while Interstellar.mainexe do
                        game:GetService("ReplicatedStorage").rEvents.raceEvent:FireServer("joinRace")
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = workspace.raceMaps.Grassland.finishPart.CFrame
                        task.wait(0.1)
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = workspace.raceMaps.Magma.finishPart.CFrame
                        task.wait(0.1)
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = workspace.raceMaps.Desert.finishPart.CFrame
                        task.wait(0.3)
                    end
                elseif game.PlaceId == 3276265788 then
                    while Interstellar.mainexe do
                        game:GetService("ReplicatedStorage").rEvents.raceEvent:FireServer("joinRace")
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = workspace.raceMaps.Speedway.finishPart.CFrame
                        task.wait(0.2)
                    end
                elseif game.PlaceId == 3232996272 then
                    while Interstellar.mainexe do
                        game:GetService("ReplicatedStorage").rEvents.raceEvent:FireServer("joinRace")
                        game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = workspace.raceMaps.Starway.finishPart.CFrame
                        task.wait(0.2)
                    end
                end
            end
        end
    })

    -- ============================================================
    -- 列表5: 传送
    -- ============================================================
    local list5 = tab:List({
        Title = "传送",
        Icon = "send",
        Opened = false,
    })

    local teleportPoints = {
        {name = "城市", pos = CFrame.new(2002.0133056640625, 1.2624330520629883, 985.2265625)},
        {name = "雪城", pos = CFrame.new(-9675.25, 59.63568115234375, 3783.50146484375)},
        {name = "熔岩城", pos = CFrame.new(-11052.4189453125, 217.59571838378906, 4898.76416015625)},
        {name = "传奇公路", pos = CFrame.new(-13095.255859375, 217.59567260742188, 5905.240234375)},
        {name = "丛林", pos = CFrame.new(-15267.9,399.0,5575.7)},
    }

    for _, tp in ipairs(teleportPoints) do
        list5:Button({
            Title = tp.name,
            Callback = function()
                game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame = tp.pos
            end
        })
    end

    -- ============================================================
    -- 列表6: 重生
    -- ============================================================
    local list6 = tab:List({
        Title = "重生",
        Icon = "rotate-ccw",
        Opened = false,
    })

    local subList3 = list6:SubList({
        Title = "模式一",
        Opened = false,
    })

    subList3:Toggle({
        Title = "自动重生",
        Default = false,
        Callback = function(state)
            Interstellar.mainexe = state
            if Interstellar.mainexe then
                spawn(function()
                    while Interstellar.mainexe do
                        pcall(function()
                            game:GetService("ReplicatedStorage").rEvents.rebirthEvent:FireServer("rebirthRequest")
                        end)
                        wait()
                    end
                end)
            end
        end
    })

    local subList4 = list6:SubList({
        Title = "模式二",
        Opened = false,
    })

    subList4:Input({
        Title = "自定义重生",
        Desc = "次数",
        Placeholder = "输入数值",
        Callback = function(value)
            Interstellar.birth = tonumber(value) or 0
        end
    })

    subList4:Input({
        Title = "重生到指定的次数",
        Desc = "输入目标重生次数",
        Placeholder = "输入数值",
        Callback = function(value)
            Interstellar.targetRebirth = tonumber(value) or 0
        end
    })

    subList4:Toggle({
        Title = "开启",
        Default = false,
        Callback = function(state)
            Interstellar.autobirth = state
            if Interstellar.autobirth then
                spawn(function()
                    local player = game:GetService("Players").LocalPlayer
                    local targetRebirths = Interstellar.targetRebirth or 0
                    while Interstellar.autobirth do
                        if player.leaderstats.Rebirths.Value >= targetRebirths then
                            Interstellar.autobirth = false
                            break
                        else
                            local success = pcall(function()
                                game:GetService("ReplicatedStorage").rEvents.rebirthEvent:FireServer("rebirthRequest")
                            end)
                            if not success then warn("重生请求失败") end
                        end
                        wait(0.1)
                    end
                end)
            end
        end
    })

    -- ============================================================
    -- 列表7: 修改
    -- ============================================================
    local list7 = tab:List({
        Title = "修改",
        Icon = "pencil",
        Opened = false,
    })

    local modifyFields = {
        {title = "修改经验值", key = "exp"},
        {title = "修改等级", key = "level"},
        {title = "修改比赛数", path = "leaderstats.Races"},
        {title = "修改环数", path = "leaderstats.Hoops"},
        {title = "修改重生", path = "leaderstats.Rebirths"},
        {title = "修改步数", path = "leaderstats.Steps"},
        {title = "修改宝石", key = "Gems"},
    }

    for _, field in ipairs(modifyFields) do
        list7:Input({
            Title = field.title,
            Placeholder = "输入数值",
            Callback = function(value)
                if value and value ~= "" and tonumber(value) then
                    local player = game:GetService("Players").LocalPlayer
                    if field.path then
                        local parts = {}
                        for part in string.gmatch(field.path, "[^%.]+") do
                            table.insert(parts, part)
                        end
                        local obj = player
                        for _, part in ipairs(parts) do
                            obj = obj[part]
                            if not obj then break end
                        end
                        if obj then obj.Value = tonumber(value) end
                    elseif field.key then
                        if player[field.key] then
                            player[field.key].Value = tonumber(value)
                        end
                    end
                end
            end
        })
    end

    -- 列表8: 宠物

        Title = "宠物",

        Title = "自动购买",
        Default = false,
        Callback = function(state)

            if state then
                spawn(function()
                    while getgenv().autoCrystal do
                        game:GetService("ReplicatedStorage").rEvents.openCrystalRemote:InvokeServer("openCrystal", selectedCrystal)

                end)
            end
        end
    })

    -- 购买宠物
    local subList6 = list8:SubList({
        Title = "购买宠物",
        Opened = false,
    })

    local petMap = {
        ["斯威夫特武士"] = "Swift Samurai",
        ["终极超速小兔"] = "Ultimate Overdrive Bunny",
        ["金色轨迹"] = "Hyperblast",
        ["彩虹轨迹"] = "Rainbow Steps",
        ["龙火轨迹"] = "Dragonfire",
        ["火焰刺猬"] = "Flaming Hedgehog",
        ["电光雷宠"] = "Electro Golem"
    }
    local petDisplayNames = {"斯威夫特武士", "终极超速小兔", "金色轨迹", "彩虹轨迹", "龙火轨迹", "火焰刺猬", "电光雷宠"}

    local BuyPetShop = ""

    subList6:Dropdown({
        Title = "选择宠物",
        Values = petDisplayNames,
        Value = "",
        Callback = function(Value)
            BuyPetShop = petMap[Value]
        end
    })

    subList6:Button({
        Title = "购买一次",
        Callback = function()
            if not BuyPetShop then return end
            local pet = game:GetService("ReplicatedStorage").cPetShopFolder:FindFirstChild(BuyPetShop)
            if pet then
                game:GetService("ReplicatedStorage").cPetShopRemote:InvokeServer(pet)
            end
        end
    })

    subList6:Toggle({
        Title = "自动购买",
        Default = false,
        Callback = function(state)
            Interstellar.petshop = state
            if Interstellar.petshop and BuyPetShop then
                spawn(function()
                    while Interstellar.petshop do
                        pcall(function()
                            local pet = game:GetService("ReplicatedStorage").cPetShopFolder:FindFirstChild(BuyPetShop)
                            if pet then
                                game:GetService("ReplicatedStorage").cPetShopRemote:InvokeServer(pet)
                            end
                        end)
                        wait(0.1)
                    end
                end)
            end
        end
    })

    -- 进化
    local subList7 = list8:SubList({
        Title = "进化",
        Opened = false,
    })

    local petEvolveMap = {
        ["斯威夫特武士"] = "Swift Samurai",
        ["火焰刺猬"] = "Flaming Hedgehog",
        ["电光雷宠"] = "Electro Golem",
        ["终极超速小兔"] = "Ultimate Overdrive Bunny"
    }

        Values = petEvolveList,
        Value = "",

        Title = "自动进化",

                    while getgenv().autoEvolve do

                            local args = {"evolvePet", EvolvePet}
                            game:GetService("ReplicatedStorage").rEvents.petEvolveEvent:FireServer(unpack(args))
                        end
                        task.wait(0.1)
                    end
                end)
            end
        end
    })

    return tab
end
